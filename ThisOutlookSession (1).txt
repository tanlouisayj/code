Option Explicit

Private WithEvents olReminders As Outlook.Reminders

' When Outlook opens: hook the reminders and catch up on today's sweep, the weekly digest and the monthly digest if they haven't run
Private Sub Application_Startup()
    Set olReminders = Application.Reminders
    SweepIfDue
    DigestIfDue
    MonthlyDigestIfDue
End Sub

' The daily/weekly calendar reminders trigger the macros
Private Sub Application_Reminder(ByVal Item As Object)
    On Error Resume Next
    Select Case Item.Subject
        Case SWEEP_SUBJECT
            DoSweep True
        Case DIGEST_SUBJECT
            DigestIfDue
        Case MONTHLY_SUBJECT
            MonthlyDigestIfDue
    End Select
End Sub

' Hide the reminder pop-ups for these three so they don't interrupt you
Private Sub olReminders_BeforeReminderShow(Cancel As Boolean)
    Dim r As Outlook.Reminder
    On Error Resume Next
    For Each r In olReminders
        If r.IsVisible Then
            If r.Caption = SWEEP_SUBJECT Or r.Caption = DIGEST_SUBJECT Or r.Caption = MONTHLY_SUBJECT Then r.Dismiss
        End If
    Next r
End Sub
