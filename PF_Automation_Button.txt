'─── SETTINGS ──────────────────────────────────────────────────────────────────

Private Function GetSetting(cellAddr As String) As String
    GetSetting = Trim(CStr(ThisWorkbook.Sheets(SETTINGS_SHEET).Range(cellAddr).Value))
End Function

Private Function WATCH_FOLDER() As String
    Dim p As String: p = GetSetting(SET_WATCH_FOLDER)
    If Right(p, 1) <> "\" Then p = p & "\"
    WATCH_FOLDER = p
End Function

Private Function QS_FILE() As String:    QS_FILE = GetSetting(SET_QS_FILE):          End Function
Private Function EMAIL_TO() As String:   EMAIL_TO = GetSetting(SET_EMAIL_TO):         End Function
Private Function EMAIL_NAME() As String: EMAIL_NAME = GetSetting(SET_EMAIL_NAME):     End Function
Private Function BASE_MONTH() As Long:   BASE_MONTH = CLng(GetSetting(SET_BASE_MONTH)): End Function
Private Function BASE_COL_IDX() As Long: BASE_COL_IDX = CLng(GetSetting(SET_BASE_COL)): End Function

Private Function OUTPUT_FOLDER() As String
    Dim p As String: p = GetSetting(SET_OUTPUT_FOLDER)
    If Right(p, 1) <> "\" Then p = p & "\"
    OUTPUT_FOLDER = p
End Function

Private Function BuildCC() As String
    Dim cc As String: cc = ""
    Dim ccList As Variant
    ccList = Array(SET_EMAIL_CC1, SET_EMAIL_CC2, SET_EMAIL_CC3, SET_EMAIL_CC4)
    Dim i As Long
    For i = 0 To UBound(ccList)
        Dim v As String: v = GetSetting(ccList(i))
        If v <> "" Then
            If cc <> "" Then cc = cc & "; "
            cc = cc & v
        End If
    Next i
    BuildCC = cc
End Function

Private Function DefaultPeriod() As String
    Dim yr As Long, mo As Long
    yr = Year(Now): mo = Month(Now) - 1
    If mo = 0 Then mo = 12: yr = yr - 1
    DefaultPeriod = Format(yr, "0000") & Format(mo, "00")
End Function

Public Sub ClearLog()
    Dim ws As Worksheet
    On Error Resume Next: Set ws = ThisWorkbook.Sheets(LOG_SHEET): On Error GoTo 0
    If ws Is Nothing Then MsgBox "No Log sheet.", vbInformation: Exit Sub
    If MsgBox("Clear all log entries?", vbYesNo + vbQuestion) = vbNo Then Exit Sub
    ws.Cells.Clear
    ws.Cells(1,1).Value="Timestamp": ws.Cells(1,2).Value="Period"
    ws.Cells(1,3).Value="Step":      ws.Cells(1,4).Value="Status"

    ws.Columns("A").ColumnWidth=22: ws.Columns("B").ColumnWidth=10
    ws.Columns("C").ColumnWidth=30: ws.Columns("D").ColumnWidth=10
    ws.Columns("E").ColumnWidth=80
    MsgBox "Log cleared.", vbInformation
End Sub

'─── LOG ───────────────────────────────────────────────────────────────────────

Private Function GetLogSheet() As Worksheet
    Dim ws As Worksheet
    On Error Resume Next: Set ws = ThisWorkbook.Sheets(LOG_SHEET): On Error GoTo 0
    If ws Is Nothing Then
        Set ws = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        ws.Name = LOG_SHEET
        ws.Cells(1,1).Value="Timestamp": ws.Cells(1,2).Value="Period"
        ws.Cells(1,3).Value="Step":      ws.Cells(1,4).Value="Status"
        ws.Cells(1,5).Value="Message"
        ws.Rows(1).Font.Bold = True
        ws.Columns("A").ColumnWidth=22: ws.Columns("B").ColumnWidth=10
        ws.Columns("C").ColumnWidth=30: ws.Columns("D").ColumnWidth=10
        ws.Columns("E").ColumnWidth=80
    End If
    Set GetLogSheet = ws
End Function

Private Sub InitLog(yyyymm As Long)
    Dim ws As Worksheet: Set ws = GetLogSheet()
    Dim r As Long: r = ws.Cells(ws.Rows.Count,1).End(xlUp).Row + 1
    ws.Cells(r,1).Value = "── NEW RUN ──": ws.Cells(r,2).Value = yyyymm
    ws.Cells(r,1).Font.Bold = True
    ws.Cells(r,1).Interior.Color = RGB(180,180,180)
    ws.Cells(r,2).Interior.Color = RGB(180,180,180)
    ws.Activate: Application.ScreenUpdating = True: DoEvents
End Sub

Private Sub WriteLog(yyyymm As Long, stepName As String, status As String, msg As String)
    Dim ws As Worksheet: Set ws = GetLogSheet()
    Dim r As Long: r = ws.Cells(ws.Rows.Count,1).End(xlUp).Row + 1
    ws.Cells(r,1).Value = Format(Now,"DD/MM/YYYY HH:MM:SS")
    ws.Cells(r,2).Value = yyyymm
    ws.Cells(r,3).Value = stepName
    ws.Cells(r,4).Value = status
    ws.Cells(r,5).Value = msg
    Select Case UCase(status)
        Case "RUNNING": ws.Cells(r,4).Interior.Color = RGB(255,255,180)
        Case "OK":      ws.Cells(r,4).Interior.Color = RGB(180,255,180)
        Case "ERROR":   ws.Cells(r,4).Interior.Color = RGB(255,180,180)
        Case "INFO":    ws.Cells(r,4).Interior.Color = RGB(180,220,255)
    End Select
    Application.ScreenUpdating = True: DoEvents
End Sub

'─── BUTTON ENTRY POINT ────────────────────────────────────────────────────────

Public Sub RunPipelineManual()
    If GetSetting(SET_WATCH_FOLDER) = "" Or GetSetting(SET_QS_FILE) = "" Or _
       GetSetting(SET_OUTPUT_FOLDER) = "" Then
        MsgBox "Fill in yellow path cells on the Control Panel first.", vbCritical
        ThisWorkbook.Sheets(SETTINGS_SHEET).Activate: Exit Sub
    End If

    Dim periodInput As String
    periodInput = InputBox("Enter the period to process:", "PF Automation", DefaultPeriod())
    If periodInput = "" Then MsgBox "Cancelled.", vbInformation: Exit Sub

    If Not IsNumeric(periodInput) Or Len(Trim(periodInput)) <> 6 Then
        MsgBox "Invalid period. Enter exactly 6 digits e.g. 202604.", vbCritical: Exit Sub
    End If

    Dim yyyymm As Long: yyyymm = CLng(periodInput)
    Dim mo As Long:     mo = yyyymm Mod 100
    If mo < 1 Or mo > 12 Then
        MsgBox "Invalid month in " & yyyymm & ". Must be 01-12.", vbCritical: Exit Sub
    End If

    Dim srcPath As String: srcPath = WATCH_FOLDER() & periodInput & ".xlsx"
    If Dir(srcPath) = "" Then
        MsgBox "File not found:" & vbNewLine & srcPath & vbNewLine & vbNewLine & _
               "Drop " & periodInput & ".xlsx into:" & vbNewLine & WATCH_FOLDER(), vbCritical
        Exit Sub
    End If

    Dim wb As Workbook
    For Each wb In Workbooks
        If InStr(1, LCase(wb.FullName), LCase("QS SQL download")) > 0 Then
            MsgBox "Close 'QS SQL download.xlsx' first.", vbCritical: Exit Sub
        End If
    Next wb

    If MsgBox("Process " & MonthLabel(yyyymm) & " (" & yyyymm & ")?" & vbNewLine & vbNewLine & _
              "To:  " & EMAIL_TO() & vbNewLine & _
              "CC:  " & BuildCC() & vbNewLine & vbNewLine & _
              "Draft email will open for review. Proceed?", _
              vbYesNo + vbQuestion, "PF Automation") = vbNo Then Exit Sub

    Call RunPipeline(srcPath, yyyymm)
End Sub

'─── MAIN PIPELINE ─────────────────────────────────────────────────────────────

Private Sub RunPipeline(srcPath As String, yyyymm As Long)
    Application.DisplayAlerts = False: Application.ScreenUpdating = False

    Call InitLog(yyyymm)
    Call WriteLog(yyyymm, "Pipeline", "INFO", "Starting " & MonthLabel(yyyymm) & " (" & yyyymm & ")")
    Call WriteLog(yyyymm, "Settings", "INFO", "To: " & EMAIL_TO() & "  CC: " & BuildCC())

    Dim srcWb As Workbook, srcWs As Worksheet
    Dim qsWb  As Workbook, qsWs  As Worksheet

    On Error GoTo PipelineError

    ' Step 1: Validate
    Call WriteLog(yyyymm, "Step 1/7 - Validate", "RUNNING", srcPath)
    Application.ScreenUpdating = False
    Set srcWb = Workbooks.Open(srcPath, ReadOnly:=True)
    Set srcWs = srcWb.Sheets(1)
    Dim errMsg As String: errMsg = RunChecks(srcWs, yyyymm)
    If errMsg <> "" Then
        Call WriteLog(yyyymm, "Step 1/7 - Validate", "ERROR", errMsg)
        srcWb.Close False
        Application.DisplayAlerts = True: Application.ScreenUpdating = True
        MsgBox "Validation failed:" & vbNewLine & errMsg, vbCritical: Exit Sub
    End If
    Call WriteLog(yyyymm, "Step 1/7 - Validate", "OK", "Passed.")

    ' Step 2: Open QS
    Call WriteLog(yyyymm, "Step 2/7 - Open QS File", "RUNNING", QS_FILE())
    Application.ScreenUpdating = False
    Set qsWb = Workbooks.Open(QS_FILE())
    Set qsWs = qsWb.Sheets(QS_DATA_TAB)
    Call WriteLog(yyyymm, "Step 2/7 - Open QS File", "OK", "Opened.")

    ' Step 3: Delete existing
    Call WriteLog(yyyymm, "Step 3/7 - Delete Rows", "RUNNING", "Removing " & yyyymm & " rows...")
    Application.ScreenUpdating = False
    Dim delCount As Long: delCount = DeleteExistingRows(qsWs, yyyymm)
    Call WriteLog(yyyymm, "Step 3/7 - Delete Rows", "OK", delCount & " rows removed.")

    ' Step 4: Copy data
    Call WriteLog(yyyymm, "Step 4/7 - Copy Data", "RUNNING", "Bulk array copy (AA, N, S as text)...")
    Application.ScreenUpdating = False
    Dim copyCount As Long: copyCount = CopyFilteredRows(srcWs, qsWs, yyyymm)
    srcWb.Close False
    Call WriteLog(yyyymm, "Step 4/7 - Copy Data", "OK", copyCount & " rows pasted.")

    ' Step 5: Formulas
    Dim lookupCol As String: lookupCol = GetLookupCol(yyyymm)
    Call WriteLog(yyyymm, "Step 5/7 - Formulas", "RUNNING", "AE/AF. Lookup col = " & lookupCol)
    Application.ScreenUpdating = False
    Call WriteFormulas(qsWs, yyyymm, lookupCol)
    Call WriteLog(yyyymm, "Step 5/7 - Formulas", "OK", "Done.")

    ' Step 6: Refresh pivots
    Call WriteLog(yyyymm, "Step 6/7 - Refresh Pivots", "RUNNING", "Refreshing (8s wait)...")
    Application.ScreenUpdating = False
    qsWb.RefreshAll
    Application.Wait Now + TimeValue("00:00:08")
    Call WriteLog(yyyymm, "Step 6/7 - Refresh Pivots", "OK", "Refreshed.")

    ' Step 7: Output
    Dim outPath As String
    outPath = OUTPUT_FOLDER() & "PF " & MonthLabel(yyyymm) & _
              " Caselist - EOP Asset Balance and Total Income by Country.xlsx"
    Call WriteLog(yyyymm, "Step 7/7 - Output File", "RUNNING", outPath)
    Application.ScreenUpdating = False

    Dim niiData() As Variant, feeData() As Variant
    Call CopyPivotsToOutput(qsWb, outPath, niiData, feeData)
    qsWb.Save: qsWb.Close
    Call WriteLog(yyyymm, "Step 7/7 - Output File", "OK", "Saved and formatted.")

    ' MoM
    Call WriteLog(yyyymm, "MoM Analysis", "RUNNING", "Scanning for YYYYMM header row...")
    Dim niiText As String, feeText As String
    niiText = GetMomContributors(niiData, "NII", yyyymm)
    feeText = GetMomContributors(feeData, "Fee", yyyymm)
    Call WriteLog(yyyymm, "MoM - NII", "INFO", niiText)
    Call WriteLog(yyyymm, "MoM - Fee", "INFO", feeText)
    Call WriteLog(yyyymm, "MoM Analysis", "OK", "Done.")

    ' Email
    Call WriteLog(yyyymm, "Email", "RUNNING", "Opening draft in Outlook...")
    Application.ScreenUpdating = True
    Call SendEmail(outPath, yyyymm, niiText, feeText)
    Call WriteLog(yyyymm, "Email", "OK", "Draft open. Review and send.")
    Call WriteLog(yyyymm, "Pipeline", "OK", "ALL COMPLETE for " & MonthLabel(yyyymm) & " ✓")

    Application.DisplayAlerts = True: Application.ScreenUpdating = True
    MsgBox "Complete for " & MonthLabel(yyyymm) & "!" & vbNewLine & vbNewLine & _
           outPath & vbNewLine & vbNewLine & "Draft email open — review and send.", _
           vbInformation, "Done"
    Exit Sub

PipelineError:
    Dim en As Long: en = Err.Number
    Dim ed As String: ed = Err.Description
    On Error Resume Next
    If Not srcWb Is Nothing Then srcWb.Close False
    If Not qsWb  Is Nothing Then qsWb.Close False
    On Error GoTo 0
    Application.CutCopyMode = False
    Call WriteLog(yyyymm, "PIPELINE ERROR", "ERROR", "Error " & en & ": " & ed)
    Application.DisplayAlerts = True: Application.ScreenUpdating = True
    MsgBox "Error " & en & ": " & ed & vbNewLine & vbNewLine & "See Log sheet.", vbCritical, "Error"
End Sub

'─── DATA CHECKS ───────────────────────────────────────────────────────────────

Private Function RunChecks(ws As Worksheet, yyyymm As Long) As String
    Dim errors As String: errors = ""
    If ws.UsedRange.Columns.Count < 27 Then
        RunChecks = "- Fewer than 27 columns in source file." & vbNewLine: Exit Function
    End If
    Dim lastRow As Long: lastRow = ws.Cells(ws.Rows.Count, COL_AA).End(xlUp).Row
    Dim found As Boolean: found = False
    Dim i As Long
    For i = 2 To lastRow
        If IsNumeric(ws.Cells(i, COL_AA).Value) Then
            If CLng(ws.Cells(i, COL_AA).Value) = yyyymm Then found = True: Exit For
        End If
    Next i
    If Not found Then errors = "- No rows where column AA = " & yyyymm & "." & vbNewLine
    RunChecks = errors
End Function

'─── DELETE EXISTING ROWS ──────────────────────────────────────────────────────

Private Function DeleteExistingRows(ws As Worksheet, yyyymm As Long) As Long
    Dim lastRow As Long: lastRow = ws.Cells(ws.Rows.Count, COL_AA).End(xlUp).Row
    Dim count As Long: count = 0
    Dim i As Long
    For i = lastRow To 2 Step -1
        If IsNumeric(ws.Cells(i, COL_AA).Value) Then
            If CLng(ws.Cells(i, COL_AA).Value) = yyyymm Then
                ws.Rows(i).Delete: count = count + 1
            End If
        End If
    Next i
    DeleteExistingRows = count
End Function

'─── BULK COPY — COLUMNS AA, N, S PRESERVED AS TEXT ───────────────────────────

Private Function CopyFilteredRows(srcWs As Worksheet, qsWs As Worksheet, yyyymm As Long) As Long
    Dim srcLastRow As Long
    srcLastRow = srcWs.Cells(srcWs.Rows.Count, COL_AA).End(xlUp).Row
    If srcLastRow < 2 Then: CopyFilteredRows = 0: Exit Function

    Dim srcData() As Variant
    srcData = srcWs.Range(srcWs.Cells(2,1), srcWs.Cells(srcLastRow,30)).Value

    Dim count As Long: count = 0
    Dim i As Long
    For i = 1 To UBound(srcData,1)
        If IsNumeric(srcData(i, COL_AA)) Then
            If CLng(srcData(i, COL_AA)) = yyyymm Then count = count + 1
        End If
    Next i
    If count = 0 Then: CopyFilteredRows = 0: Exit Function

    Dim outData() As Variant
    ReDim outData(1 To count, 1 To 30)
    Dim destIdx As Long: destIdx = 0
    Dim j As Long
    For i = 1 To UBound(srcData,1)
        If IsNumeric(srcData(i, COL_AA)) Then
            If CLng(srcData(i, COL_AA)) = yyyymm Then
                destIdx = destIdx + 1
                For j = 1 To 30
                    ' Preserve AA, N, S as text strings
                    If j = COL_AA Or j = COL_N Or j = COL_S Then
                        outData(destIdx, j) = CStr(srcData(i, j))
                    Else
                        outData(destIdx, j) = srcData(i, j)
                    End If
                Next j
            End If
        End If
    Next i

    ' Pre-format AA, N, S as text before writing
    Dim destRow As Long
    destRow = qsWs.Cells(qsWs.Rows.Count,1).End(xlUp).Row + 1
    qsWs.Cells(destRow, COL_AA).Resize(count,1).NumberFormat = "@"
    qsWs.Cells(destRow, COL_N).Resize(count,1).NumberFormat  = "@"
    qsWs.Cells(destRow, COL_S).Resize(count,1).NumberFormat  = "@"

    ' Write all at once
    qsWs.Cells(destRow,1).Resize(count,30).Value = outData
    CopyFilteredRows = count
End Function

'─── WRITE AE / AF FORMULAS ────────────────────────────────────────────────────

Private Sub WriteFormulas(ws As Worksheet, yyyymm As Long, lookupCol As String)
    Dim lastRow As Long: lastRow = ws.Cells(ws.Rows.Count,1).End(xlUp).Row
    Dim i As Long
    For i = 2 To lastRow
        If IsNumeric(ws.Cells(i, COL_AA).Value) Then
            If CLng(ws.Cells(i, COL_AA).Value) = yyyymm Then
                ws.Cells(i, COL_AE).Formula = "=F" & i & "&E" & i & "&N" & i
                ws.Cells(i, COL_AF).Formula = _
                    "=IF(ISNUMBER(SEARCH(""card"",Q" & i & ")),""Card""," & _
                    "IFERROR(VLOOKUP(AE" & i & ",'" & CHECK_DUP_TAB & "'!" & _
                    lookupCol & ":" & lookupCol & ",1,0),""*No""))"
            End If
        End If
    Next i
End Sub

'─── COPY PIVOTS TO OUTPUT ─────────────────────────────────────────────────────

Private Sub CopyPivotsToOutput(qsWb As Workbook, outPath As String, _
                                niiData() As Variant, feeData() As Variant)
    Dim pivotWs As Worksheet
    Set pivotWs = qsWb.Sheets(PIVOT_TAB)

    Dim outWb As Workbook
    Set outWb = Workbooks.Add
    Do While outWb.Sheets.Count > 1
        outWb.Sheets(outWb.Sheets.Count).Delete
    Loop

    Dim pivotCells(1 To 5) As String
    Dim tabNames(1 To 5)   As String
    pivotCells(1) = "A1":  tabNames(1) = "EOP Asset Balance"
    pivotCells(2) = "S1":  tabNames(2) = "Total Income"
    pivotCells(3) = "AP1": tabNames(3) = "NII"
    pivotCells(4) = "BH1": tabNames(4) = "Fee"
    pivotCells(5) = "BZ1": tabNames(5) = "Other income"

    Dim idx As Long
    For idx = 1 To 5
        Dim startCell As Range
        Set startCell = pivotWs.Range(pivotCells(idx))

        Dim pvRange As Range
        Dim pt As PivotTable
        Dim found As Boolean: found = False

        For Each pt In pivotWs.PivotTables
            If pt.TableRange2.Cells(1,1).Address = startCell.Address Then
                Set pvRange = pt.TableRange2: found = True: Exit For
            End If
        Next pt

        If Not found Then
            For Each pt In pivotWs.PivotTables
                If Not Intersect(pt.TableRange2, startCell) Is Nothing Then
                    Set pvRange = pt.TableRange2: found = True: Exit For
                End If
            Next pt
        End If

        If Not found Then Set pvRange = startCell.CurrentRegion

        ' Capture values for MoM BEFORE formatting
        Dim pvData() As Variant: pvData = pvRange.Value
        If idx = 3 Then niiData = pvData
        If idx = 4 Then feeData = pvData

        Dim outWs As Worksheet
        If idx = 1 Then
            Set outWs = outWb.Sheets(1)
            outWs.Name = tabNames(idx)
        Else
            Set outWs = outWb.Sheets.Add(After:=outWb.Sheets(outWb.Sheets.Count))
            outWs.Name = tabNames(idx)
        End If

        pvRange.Copy
        outWs.Range("A1").PasteSpecial xlPasteValues
        outWs.Range("A1").PasteSpecial xlPasteFormats
        Application.CutCopyMode = False

        Call FormatOutputSheet(outWs)
    Next idx

    outWb.SaveAs outPath, xlOpenXMLWorkbook
    outWb.Close
End Sub

'─── FORMAT OUTPUT SHEET ───────────────────────────────────────────────────────

Private Sub FormatOutputSheet(ws As Worksheet)
    ' Delete top 3 rows (Level 3.5 Product Desc, Exclude, blank)
    ws.Rows("1:3").Delete

    Dim lastRow As Long: lastRow = ws.Cells(ws.Rows.Count,1).End(xlUp).Row
    Dim lastCol As Long: lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    If lastCol < 1 Then lastCol = 1

    ' Row 1 (was row 4): light blue header — RGB(192,230,245) = #C0E6F5
    With ws.Range(ws.Cells(1,1), ws.Cells(1, lastCol))
        .Font.Bold = True
        .Font.Color = RGB(0, 0, 0)         ' black font
        .Interior.Color = RGB(192, 230, 245) ' light blue #C0E6F5
    End With

    ' Row 2 (was row 5): column header — red RGB(192,0,0), white font
    With ws.Range(ws.Cells(2,1), ws.Cells(2, lastCol))
        .Font.Bold = True
        .Font.Color = RGB(255, 255, 255)   ' white font
        .Interior.Color = RGB(192, 0, 0)   ' red
    End With

    ' Bold columns A and B throughout (rows 1 to lastRow)
    ws.Range(ws.Cells(1,1), ws.Cells(lastRow,2)).Font.Bold = True

    ' Bold all rows containing "total" (from row 3 onwards)
    Dim r As Long, c As Long
    For r = 3 To lastRow
        For c = 1 To lastCol
            If Not IsEmpty(ws.Cells(r,c).Value) Then
                If InStr(1, LCase(CStr(ws.Cells(r,c).Value)), "total") > 0 Then
                    ws.Rows(r).Font.Bold = True
                    Exit For
                End If
            End If
        Next c
    Next r

    ' Autofit all columns, no wrap
    ws.Cells.WrapText = False
    Dim col As Long
    For col = 1 To lastCol
        ws.Columns(col).AutoFit
    Next col
End Sub

'─── MOM ANALYSIS — SCANS ALL ROWS FOR YYYYMM HEADER ──────────────────────────

Private Function GetMomContributors(pvData() As Variant, label As String, yyyymm As Long) As String
    On Error GoTo ErrHandler
    If Not IsArray(pvData) Then
        GetMomContributors = label & ": no data.": Exit Function
    End If

    Dim prevMm As Long: prevMm = PrevYYYYMM(yyyymm)

    ' Scan ALL rows to find the header row containing YYYYMM values
    ' (handles pivot structures where months aren't in row 1)
    Dim headerRow As Long: headerRow = 0
    Dim r As Long, c As Long
    For r = 1 To UBound(pvData,1)
        For c = 1 To UBound(pvData,2)
            If IsNumeric(pvData(r,c)) Then
                Dim testVal As Long
                testVal = CLng(pvData(r,c))
                ' Valid YYYYMM range: 200001 to 209912
                If testVal >= 200001 And testVal <= 209912 Then
                    Dim mo As Long: mo = testVal Mod 100
                    If mo >= 1 And mo <= 12 Then
                        headerRow = r
                        Exit For
                    End If
                End If
            End If
        Next c
        If headerRow > 0 Then Exit For
    Next r

    If headerRow = 0 Then
        GetMomContributors = label & ": could not find YYYYMM header row in pivot data."
        Exit Function
    End If

    ' Find current and previous month columns
    Dim currCol As Long, prevCol As Long
    currCol = 0: prevCol = 0
    For c = 1 To UBound(pvData,2)
        If IsNumeric(pvData(headerRow, c)) Then
            If CLng(pvData(headerRow, c)) = yyyymm Then currCol = c
            If CLng(pvData(headerRow, c)) = prevMm  Then prevCol = c
        End If
    Next c

    If currCol = 0 Or prevCol = 0 Then
        GetMomContributors = label & ": columns not found — " & yyyymm & _
                             " (col " & currCol & ") or " & prevMm & " (col " & prevCol & ")" & _
                             " in header row " & headerRow
        Exit Function
    End If

    ' Collect movements from rows below the header
    Dim names() As String, moves() As Double, count As Long: count = 0
    For r = headerRow + 1 To UBound(pvData,1)
        Dim rowLabel As String: rowLabel = ""
        If Not IsEmpty(pvData(r,1)) Then rowLabel = Trim(CStr(pvData(r,1)))
        If rowLabel = "" Or LCase(rowLabel) = "grand total" Then GoTo NextRow
        Dim currVal As Double, prevVal As Double
        currVal = 0: prevVal = 0
        If IsNumeric(pvData(r, currCol)) Then currVal = CDbl(pvData(r, currCol))
        If IsNumeric(pvData(r, prevCol)) Then prevVal = CDbl(pvData(r, prevCol))
        Dim mv As Double: mv = currVal - prevVal
        If mv <> 0 Then
            count = count + 1
            ReDim Preserve names(1 To count): ReDim Preserve moves(1 To count)
            names(count) = rowLabel: moves(count) = mv
        End If
NextRow:
    Next r

    If count = 0 Then
        GetMomContributors = label & ": no MoM movement detected.": Exit Function
    End If

    ' Sort by absolute value descending
    Dim j As Long, tn As String, tm As Double
    For r = 1 To count - 1
        For j = 1 To count - r
            If Abs(moves(j)) < Abs(moves(j+1)) Then
                tn = names(j): tm = moves(j)
                names(j) = names(j+1): moves(j) = moves(j+1)
                names(j+1) = tn: moves(j+1) = tm
            End If
        Next j
    Next r

    Dim totalMove As Double, absTotal As Double
    totalMove = 0: absTotal = 0
    For r = 1 To count
        totalMove = totalMove + moves(r)
        absTotal  = absTotal  + Abs(moves(r))
    Next r

    Dim result As String
    result = label & " MoM (" & MonthLabel(prevMm) & " to " & MonthLabel(yyyymm) & "): " & _
             Format(Abs(totalMove), "#,##0") & IIf(totalMove >= 0, " up", " down") & vbNewLine & _
             "Key contributors (>=80%):" & vbNewLine

    Dim cumulative As Double: cumulative = 0
    For r = 1 To count
        Dim pct As Double
        pct = IIf(totalMove <> 0, moves(r) / totalMove * 100, 0)
        result = result & "  " & IIf(moves(r) >= 0, Chr(9650), Chr(9660)) & " " & _
                 names(r) & ": " & Format(moves(r), "+#,##0;-#,##0") & _
                 " (" & Format(pct, "+0.0;-0.0") & "%)" & vbNewLine
        cumulative = cumulative + Abs(moves(r))
        If absTotal > 0 And cumulative / absTotal >= 0.8 Then Exit For
    Next r

    GetMomContributors = result
    Exit Function
ErrHandler:
    GetMomContributors = label & ": error - " & Err.Description
End Function

'─── HELPERS ───────────────────────────────────────────────────────────────────

Private Function GetLookupCol(yyyymm As Long) As String
    Dim diff As Long, colIdx As Long
    diff   = ((yyyymm \ 100) - (BASE_MONTH() \ 100)) * 12 + _
             ((yyyymm Mod 100) - (BASE_MONTH() Mod 100))
    colIdx = BASE_COL_IDX() + diff
    If colIdx <= 26 Then
        GetLookupCol = Chr(64 + colIdx)
    Else
        GetLookupCol = Chr(64 + (colIdx-1) \ 26) & Chr(65 + (colIdx-1) Mod 26)
    End If
End Function

Private Function PrevYYYYMM(yyyymm As Long) As Long
    Dim yr As Long, mo As Long
    yr = yyyymm \ 100: mo = yyyymm Mod 100
    If mo = 1 Then PrevYYYYMM = (yr-1)*100+12 Else PrevYYYYMM = yr*100+(mo-1)
End Function

Private Function MonthLabel(yyyymm As Long) As String
    Dim m(1 To 12) As String
    m(1)="Jan": m(2)="Feb": m(3)="Mar": m(4)="Apr":  m(5)="May":  m(6)="Jun"
    m(7)="Jul": m(8)="Aug": m(9)="Sep": m(10)="Oct": m(11)="Nov": m(12)="Dec"
    MonthLabel = m(yyyymm Mod 100) & " " & Right(CStr(yyyymm \ 100), 2)
End Function
