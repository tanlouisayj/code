'─── LOG HELPERS ──────────────────────────────────────────────────────────────

Private Sub InitLog(yyyymm As Long)
    Dim ws As Worksheet

    ' Create Log sheet if it doesn't exist
    On Error Resume Next
    Set ws = ThisWorkbook.Sheets(LOG_SHEET)
    On Error GoTo 0

    If ws Is Nothing Then
        Set ws = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        ws.Name = LOG_SHEET
    End If

    ' Set up headers if sheet is empty
    If ws.Cells(1, 1).Value = "" Then
        ws.Cells(1, 1).Value = "Timestamp"
        ws.Cells(1, 2).Value = "Period"
        ws.Cells(1, 3).Value = "Step"
        ws.Cells(1, 4).Value = "Status"
        ws.Cells(1, 5).Value = "Message"
        ws.Rows(1).Font.Bold = True
        ws.Columns("A").ColumnWidth = 20
        ws.Columns("B").ColumnWidth = 10
        ws.Columns("C").ColumnWidth = 35
        ws.Columns("D").ColumnWidth = 10
        ws.Columns("E").ColumnWidth = 60
    End If

    ' Add separator row for new run
    Dim nextRow As Long
    nextRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row + 1
    ws.Cells(nextRow, 1).Value = "── NEW RUN ──"
    ws.Cells(nextRow, 2).Value = yyyymm
    ws.Cells(nextRow, 1).Font.Bold = True
    ws.Cells(nextRow, 1).Interior.Color = RGB(200, 200, 200)
    ws.Cells(nextRow, 2).Interior.Color = RGB(200, 200, 200)

    ' Switch to log sheet so user can watch
    ws.Activate
    Application.ScreenUpdating = True
End Sub

Private Sub WriteLog(yyyymm As Long, stepName As String, status As String, msg As String)
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = ThisWorkbook.Sheets(LOG_SHEET)
    On Error GoTo 0
    If ws Is Nothing Then Exit Sub

    Dim nextRow As Long
    nextRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row + 1

    ws.Cells(nextRow, 1).Value = Format(Now, "DD/MM/YYYY HH:MM:SS")
    ws.Cells(nextRow, 2).Value = yyyymm
    ws.Cells(nextRow, 3).Value = stepName
    ws.Cells(nextRow, 4).Value = status
    ws.Cells(nextRow, 5).Value = msg

    ' Colour by status
    Select Case UCase(status)
        Case "RUNNING"
            ws.Cells(nextRow, 4).Interior.Color = RGB(255, 255, 180)  ' Yellow
        Case "OK"
            ws.Cells(nextRow, 4).Interior.Color = RGB(180, 255, 180)  ' Green
        Case "ERROR"
            ws.Cells(nextRow, 4).Interior.Color = RGB(255, 180, 180)  ' Red
        Case "INFO"
            ws.Cells(nextRow, 4).Interior.Color = RGB(180, 220, 255)  ' Blue
    End Select

    ' Scroll to latest row
    ws.Cells(nextRow, 1).Select
    Application.ScreenUpdating = True
    DoEvents
End Sub

'─── BUTTON ENTRY POINT ───────────────────────────────────────────────────────

Public Sub RunPipelineManual()
    Dim periodInput As String
    periodInput = InputBox("Enter the period to process (e.g. 202605):", "PF Automation", Format(Now, "YYYYMM"))

    If periodInput = "" Then
        MsgBox "Cancelled.", vbInformation
        Exit Sub
    End If

    If Not IsNumeric(periodInput) Or Len(periodInput) <> 6 Then
        MsgBox "Invalid period. Please enter 6 digits e.g. 202605.", vbCritical
        Exit Sub
    End If

    Dim yyyymm As Long
    yyyymm = CLng(periodInput)

    Dim mo As Long
    mo = yyyymm Mod 100
    If mo < 1 Or mo > 12 Then
        MsgBox "Invalid month in period " & yyyymm & ". Month must be 01-12.", vbCritical
        Exit Sub
    End If

    Dim srcPath As String
    srcPath = WATCH_FOLDER & periodInput & ".xlsx"

    If Dir(srcPath) = "" Then
        MsgBox "Source file not found:" & vbNewLine & srcPath & vbNewLine & vbNewLine & _
               "Please drop " & periodInput & ".xlsx into the Automation folder first.", vbCritical
        Exit Sub
    End If

    Dim confirm As Integer
    confirm = MsgBox("Ready to process period " & yyyymm & "." & vbNewLine & vbNewLine & _
                     "Source:  " & srcPath & vbNewLine & _
                     "QS File: " & QS_FILE & vbNewLine & _
                     "Output:  " & OUTPUT_FOLDER & "PF " & MonthLabel(yyyymm) & " Caselist..." & vbNewLine & vbNewLine & _
                     "Draft email will open for review before sending." & vbNewLine & vbNewLine & _
                     "Proceed?", vbYesNo + vbQuestion, "PF Automation")

    If confirm = vbNo Then Exit Sub

    Call RunPipeline(srcPath, yyyymm)
End Sub

'─── MAIN PIPELINE ────────────────────────────────────────────────────────────

Private Sub RunPipeline(srcPath As String, yyyymm As Long)
    Application.DisplayAlerts = False

    Call InitLog(yyyymm)
    Call WriteLog(yyyymm, "Pipeline", "INFO", "Starting pipeline for period " & yyyymm)

    On Error GoTo PipelineError

    '── Step 1: Validate source file ──────────────────────────────────────────
    Call WriteLog(yyyymm, "Step 1/7 - Validate", "RUNNING", "Opening source file: " & srcPath)
    Dim srcWb As Workbook
    Set srcWb = Workbooks.Open(srcPath, ReadOnly:=True)
    Dim srcWs As Worksheet
    Set srcWs = srcWb.Sheets(1)

    Dim errMsg As String
    errMsg = RunChecks(srcWs, yyyymm)
    If errMsg <> "" Then
        Call WriteLog(yyyymm, "Step 1/7 - Validate", "ERROR", "Checks failed: " & errMsg)
        srcWb.Close False
        MsgBox "Data checks failed:" & vbNewLine & errMsg, vbCritical, "Validation Failed"
        GoTo Cleanup
    End If
    Call WriteLog(yyyymm, "Step 1/7 - Validate", "OK", "Source file validated. Columns OK, period data found.")

    '── Step 2: Open QS SQL download ──────────────────────────────────────────
    Call WriteLog(yyyymm, "Step 2/7 - Open QS File", "RUNNING", "Opening: " & QS_FILE)
    Dim qsWb As Workbook
    Set qsWb = Workbooks.Open(QS_FILE)
    Dim qsWs As Worksheet
    Set qsWs = qsWb.Sheets(QS_DATA_TAB)
    Call WriteLog(yyyymm, "Step 2/7 - Open QS File", "OK", "QS SQL download.xlsx opened successfully.")

    '── Step 3: Delete existing rows ──────────────────────────────────────────
    Call WriteLog(yyyymm, "Step 3/7 - Delete Rows", "RUNNING", "Removing existing rows for period " & yyyymm)
    Dim deletedCount As Long
    deletedCount = DeleteExistingRows(qsWs, yyyymm)
    Call WriteLog(yyyymm, "Step 3/7 - Delete Rows", "OK", deletedCount & " existing rows removed.")

    '── Step 4: Copy filtered rows ────────────────────────────────────────────
    Call WriteLog(yyyymm, "Step 4/7 - Copy Data", "RUNNING", "Copying filtered rows from source file...")
    Dim copiedCount As Long
    copiedCount = CopyFilteredRows(srcWs, qsWs, yyyymm)
    srcWb.Close False
    Call WriteLog(yyyymm, "Step 4/7 - Copy Data", "OK", copiedCount & " rows copied into QS data tab.")

    '── Step 5: Write formulas ────────────────────────────────────────────────
    Dim lookupCol As String
    lookupCol = GetLookupCol(yyyymm)
    Call WriteLog(yyyymm, "Step 5/7 - Write Formulas", "RUNNING", "Writing AE/AF formulas. Lookup column: " & lookupCol)
    Call WriteFormulas(qsWs, yyyymm)
    Call WriteLog(yyyymm, "Step 5/7 - Write Formulas", "OK", "Formulas written for " & copiedCount & " rows.")

    '── Step 6: Refresh pivots ────────────────────────────────────────────────
    Call WriteLog(yyyymm, "Step 6/7 - Refresh Pivots", "RUNNING", "Refreshing all pivots in Caselist Pivots tab...")
    qsWb.RefreshAll
    Application.Wait Now + TimeValue("00:00:06")
    Call WriteLog(yyyymm, "Step 6/7 - Refresh Pivots", "OK", "All pivots refreshed successfully.")

    '── Step 7: Copy pivots to output ─────────────────────────────────────────
    Dim outPath As String
    outPath = OUTPUT_FOLDER & "PF " & MonthLabel(yyyymm) & _
              " Caselist - EOP Asset Balance and Total Income by Country.xlsx"
    Call WriteLog(yyyymm, "Step 7/7 - Output File", "RUNNING", "Creating output file: " & outPath)

    Dim niiData() As Variant
    Dim feeData()  As Variant
    Call CopyPivotsToOutput(qsWb, outPath, niiData, feeData)

    qsWb.Save
    qsWb.Close
    Call WriteLog(yyyymm, "Step 7/7 - Output File", "OK", "Output file saved. QS SQL download.xlsx saved and closed.")

    '── MoM analysis ──────────────────────────────────────────────────────────
    Call WriteLog(yyyymm, "MoM Analysis", "RUNNING", "Calculating NII and Fee MoM contributors...")
    Dim niiText As String, feeText As String
    niiText = GetMomContributors(niiData, "NII", yyyymm)
    feeText = GetMomContributors(feeData, "Fee", yyyymm)
    Call WriteLog(yyyymm, "MoM Analysis", "OK", "MoM analysis complete.")
    Call WriteLog(yyyymm, "MoM - NII", "INFO", niiText)
    Call WriteLog(yyyymm, "MoM - Fee", "INFO", feeText)

    '── Email draft ───────────────────────────────────────────────────────────
    Call WriteLog(yyyymm, "Email", "RUNNING", "Preparing draft email for review...")
    Call SendEmail(outPath, yyyymm, niiText, feeText)
    Call WriteLog(yyyymm, "Email", "OK", "Draft email opened in Outlook. Please review and send manually.")

    '── Done ──────────────────────────────────────────────────────────────────
    Call WriteLog(yyyymm, "Pipeline", "OK", "ALL STEPS COMPLETE for " & MonthLabel(yyyymm) & " ✓")

    Application.DisplayAlerts = True
    MsgBox "Done! Pipeline complete for " & MonthLabel(yyyymm) & "." & vbNewLine & vbNewLine & _
           "Output: " & outPath & vbNewLine & _
           "Draft email is open in Outlook — please review and send.", _
           vbInformation, "PF Automation Complete"
    Exit Sub

PipelineError:
    Call WriteLog(yyyymm, "PIPELINE ERROR", "ERROR", "Error " & Err.Number & ": " & Err.Description)
    Application.DisplayAlerts = True
    MsgBox "Pipeline error:" & vbNewLine & vbNewLine & _
           "Error " & Err.Number & ": " & Err.Description & vbNewLine & vbNewLine & _
           "Check the Log sheet for details.", vbCritical, "Pipeline Error"

Cleanup:
    Application.DisplayAlerts = True
End Sub

'─── DATA CHECKS ──────────────────────────────────────────────────────────────

Private Function RunChecks(ws As Worksheet, yyyymm As Long) As String
    Dim errors As String
    errors = ""

    If ws.UsedRange.Columns.Count < 30 Then
        errors = errors & "- Source file has fewer than 30 columns." & vbNewLine
    End If

    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, COL_AA).End(xlUp).Row

    Dim found As Boolean
    found = False
    Dim i As Long
    For i = 2 To lastRow
        If IsNumeric(ws.Cells(i, COL_AA).Value) Then
            If CLng(ws.Cells(i, COL_AA).Value) = yyyymm Then
                found = True
                Exit For
            End If
        End If
    Next i

    If Not found Then
        errors = errors & "- No rows where column AA = " & yyyymm & " found in source file." & vbNewLine
    End If

    RunChecks = errors
End Function

'─── DELETE EXISTING ROWS ─────────────────────────────────────────────────────

Private Function DeleteExistingRows(ws As Worksheet, yyyymm As Long) As Long
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, COL_AA).End(xlUp).Row
    Dim count As Long
    count = 0

    Dim i As Long
    For i = lastRow To 2 Step -1
        If IsNumeric(ws.Cells(i, COL_AA).Value) Then
            If CLng(ws.Cells(i, COL_AA).Value) = yyyymm Then
                ws.Rows(i).Delete
                count = count + 1
            End If
        End If
    Next i
    DeleteExistingRows = count
End Function

'─── COPY FILTERED ROWS ───────────────────────────────────────────────────────

Private Function CopyFilteredRows(srcWs As Worksheet, qsWs As Worksheet, yyyymm As Long) As Long
    Dim srcLastRow As Long
    srcLastRow = srcWs.Cells(srcWs.Rows.Count, COL_AA).End(xlUp).Row

    Dim destRow As Long
    destRow = qsWs.Cells(qsWs.Rows.Count, 1).End(xlUp).Row + 1

    Dim count As Long
    count = 0
    Dim i As Long
    For i = 2 To srcLastRow
        If IsNumeric(srcWs.Cells(i, COL_AA).Value) Then
            If CLng(srcWs.Cells(i, COL_AA).Value) = yyyymm Then
                srcWs.Range(srcWs.Cells(i, 1), srcWs.Cells(i, 30)).Copy
                qsWs.Cells(destRow, 1).PasteSpecial xlPasteValues
                destRow = destRow + 1
                count = count + 1
            End If
        End If
    Next i
    Application.CutCopyMode = False
    CopyFilteredRows = count
End Function

'─── WRITE AE / AF FORMULAS ───────────────────────────────────────────────────

Private Sub WriteFormulas(ws As Worksheet, yyyymm As Long)
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row

    Dim lookupCol As String
    lookupCol = GetLookupCol(yyyymm)

    Dim i As Long
    For i = 2 To lastRow
        If IsNumeric(ws.Cells(i, COL_AA).Value) Then
            If CLng(ws.Cells(i, COL_AA).Value) = yyyymm Then
                ws.Cells(i, COL_AE).Formula = "=F" & i & "&E" & i & "&N" & i
                ws.Cells(i, COL_AF).Formula = _
                    "=IF(ISNUMBER(SEARCH(""card"",Q" & i & ")),""Card""," & _
                    "IFERROR(VLOOKUP(AE" & i & ",'" & CHECK_DUP_TAB & "'!" & _
                    lookupCol & ":" & lookupCol & ",1,0),""*No""))"
            End If
        End If
    Next i
End Sub

'─── COPY PIVOTS TO OUTPUT ────────────────────────────────────────────────────

Private Sub CopyPivotsToOutput(qsWb As Workbook, outPath As String, _
                                niiData() As Variant, feeData() As Variant)
    Dim pivotWs As Worksheet
    Set pivotWs = qsWb.Sheets(PIVOT_TAB)

    Dim outWb As Workbook
    Set outWb = Workbooks.Add
    Do While outWb.Sheets.Count > 1
        outWb.Sheets(outWb.Sheets.Count).Delete
    Loop

    Dim pivotCells(1 To 5) As String
    Dim tabNames(1 To 5)   As String
    pivotCells(1) = "A1":  tabNames(1) = "EOP Asset Balance"
    pivotCells(2) = "S1":  tabNames(2) = "Total Income"
    pivotCells(3) = "AP1": tabNames(3) = "NII"
    pivotCells(4) = "BH1": tabNames(4) = "Fee"
    pivotCells(5) = "BZ1": tabNames(5) = "Other income"

    Dim idx As Long
    For idx = 1 To 5
        Dim pvData() As Variant
        pvData = pivotWs.Range(pivotCells(idx)).CurrentRegion.Value

        If idx = 3 Then niiData = pvData
        If idx = 4 Then feeData = pvData

        Dim outWs As Worksheet
        If idx = 1 Then
            Set outWs = outWb.Sheets(1)
            outWs.Name = tabNames(idx)
        Else
            Set outWs = outWb.Sheets.Add(After:=outWb.Sheets(outWb.Sheets.Count))
            outWs.Name = tabNames(idx)
        End If

        outWs.Range("A1").Resize(UBound(pvData, 1), UBound(pvData, 2)).Value = pvData
    Next idx

    outWb.SaveAs outPath, xlOpenXMLWorkbook
    outWb.Close
End Sub

'─── MOM ANALYSIS ─────────────────────────────────────────────────────────────

Private Function GetMomContributors(pvData() As Variant, label As String, yyyymm As Long) As String
    On Error GoTo ErrHandler

    If Not IsArray(pvData) Then
        GetMomContributors = label & ": no data available.": Exit Function
    End If

    Dim prevMm As Long
    prevMm = PrevYYYYMM(yyyymm)

    Dim currCol As Long, prevCol As Long
    currCol = 0: prevCol = 0

    Dim c As Long
    For c = 1 To UBound(pvData, 2)
        If IsNumeric(pvData(1, c)) Then
            If CLng(pvData(1, c)) = yyyymm Then currCol = c
            If CLng(pvData(1, c)) = prevMm  Then prevCol = c
        End If
    Next c

    If currCol = 0 Or prevCol = 0 Then
        GetMomContributors = label & ": could not locate month columns.": Exit Function
    End If

    Dim names() As String
    Dim moves() As Double
    Dim count   As Long
    count = 0

    Dim r As Long
    For r = 2 To UBound(pvData, 1)
        Dim rowLabel As String
        rowLabel = ""
        If Not IsEmpty(pvData(r, 1)) Then rowLabel = Trim(CStr(pvData(r, 1)))
        If rowLabel = "" Or LCase(rowLabel) = "grand total" Then GoTo NextRow

        Dim currVal As Double, prevVal As Double
        currVal = 0: prevVal = 0
        If IsNumeric(pvData(r, currCol)) Then currVal = CDbl(pvData(r, currCol))
        If IsNumeric(pvData(r, prevCol)) Then prevVal = CDbl(pvData(r, prevCol))

        Dim movement As Double
        movement = currVal - prevVal
        If movement <> 0 Then
            count = count + 1
            ReDim Preserve names(1 To count)
            ReDim Preserve moves(1 To count)
            names(count) = rowLabel
            moves(count) = movement
        End If
NextRow:
    Next r

    If count = 0 Then
        GetMomContributors = label & ": no MoM movement detected.": Exit Function
    End If

    Dim j As Long
    Dim tempN As String, tempM As Double
    For r = 1 To count - 1
        For j = 1 To count - r
            If Abs(moves(j)) < Abs(moves(j + 1)) Then
                tempN = names(j): tempM = moves(j)
                names(j) = names(j + 1): moves(j) = moves(j + 1)
                names(j + 1) = tempN: moves(j + 1) = tempM
            End If
        Next j
    Next r

    Dim totalMove As Double, absTotal As Double
    totalMove = 0: absTotal = 0
    For r = 1 To count
        totalMove = totalMove + moves(r)
        absTotal  = absTotal  + Abs(moves(r))
    Next r

    Dim direction As String
    direction = IIf(totalMove >= 0, "up", "down")

    Dim lines As String
    lines = label & " MoM (" & MonthLabel(prevMm) & " to " & MonthLabel(yyyymm) & "): " & _
            Format(Abs(totalMove), "#,##0") & " " & direction & vbNewLine & _
            "Key contributors (>=80% of movement):" & vbNewLine

    Dim cumulative As Double
    cumulative = 0
    For r = 1 To count
        Dim pct As Double
        pct = IIf(totalMove <> 0, moves(r) / totalMove * 100, 0)
        Dim arrow As String
        arrow = IIf(moves(r) >= 0, Chr(9650), Chr(9660))
        lines = lines & "  " & arrow & " " & names(r) & ": " & _
                Format(moves(r), "+#,##0;-#,##0") & " (" & Format(pct, "+0.0;-0.0") & "%)" & vbNewLine
        cumulative = cumulative + Abs(moves(r))
        If absTotal > 0 And cumulative / absTotal >= 0.8 Then Exit For
    Next r

    GetMomContributors = lines
    Exit Function
ErrHandler:
    GetMomContributors = label & ": error - " & Err.Description
End Function

'─── SEND EMAIL ───────────────────────────────────────────────────────────────

Private Sub SendEmail(attachPath As String, yyyymm As Long, niiText As String, feeText As String)
    Dim olApp  As Object
    Dim olMail As Object
    Set olApp  = CreateObject("Outlook.Application")
    Set olMail = olApp.CreateItem(0)

    Dim body As String
    body = "Hi Steve," & vbNewLine & vbNewLine & _
           "Please find attached the updated PF Caselist - " & _
           "EOP Asset Balance and Total Income by Country." & _
           vbNewLine & vbNewLine & _
           niiText & vbNewLine & _
           feeText & vbNewLine & _
           "Best regards," & vbNewLine & EMAIL_NAME

    With olMail
        .To      = EMAIL_TO
        .CC      = EMAIL_CC
        .Subject = "PF " & MonthLabel(yyyymm) & _
                   " Caselist - EOP Asset Balance and Total Income by Country"
        .Body    = body
        .Attachments.Add attachPath
        .Display
    End With
End Sub

'─── HELPERS ──────────────────────────────────────────────────────────────────

Private Function GetLookupCol(yyyymm As Long) As String
    Dim diff   As Long
    Dim colIdx As Long
    diff   = ((yyyymm \ 100) - (BASE_MONTH \ 100)) * 12 + _
             ((yyyymm Mod 100) - (BASE_MONTH Mod 100))
    colIdx = BASE_COL_IDX + diff
    If colIdx <= 26 Then
        GetLookupCol = Chr(64 + colIdx)
    Else
        GetLookupCol = Chr(64 + (colIdx - 1) \ 26) & Chr(65 + (colIdx - 1) Mod 26)
    End If
End Function

Private Function PrevYYYYMM(yyyymm As Long) As Long
    Dim yr As Long, mo As Long
    yr = yyyymm \ 100: mo = yyyymm Mod 100
    If mo = 1 Then PrevYYYYMM = (yr - 1) * 100 + 12 _
    Else PrevYYYYMM = yr * 100 + (mo - 1)
End Function

Private Function MonthLabel(yyyymm As Long) As String
    Dim mNames(1 To 12) As String
    mNames(1) = "Jan": mNames(2) = "Feb": mNames(3)  = "Mar"
    mNames(4) = "Apr": mNames(5) = "May": mNames(6)  = "Jun"
    mNames(7) = "Jul": mNames(8) = "Aug": mNames(9)  = "Sep"
    mNames(10) = "Oct": mNames(11) = "Nov": mNames(12) = "Dec"
    MonthLabel = mNames(yyyymm Mod 100) & " " & Right(CStr(yyyymm \ 100), 2)
End Function
