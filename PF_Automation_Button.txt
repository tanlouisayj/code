'─── LOG FUNCTIONS ─────────────────────────────────────────────────────────────

Private Function GetLogSheet() As Worksheet
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = ThisWorkbook.Sheets(LOG_SHEET)
    On Error GoTo 0

    If ws Is Nothing Then
        Set ws = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        ws.Name = LOG_SHEET
        ws.Cells(1, 1).Value = "Timestamp"
        ws.Cells(1, 2).Value = "Period"
        ws.Cells(1, 3).Value = "Step"
        ws.Cells(1, 4).Value = "Status"
        ws.Cells(1, 5).Value = "Message"
        ws.Rows(1).Font.Bold = True
        ws.Columns("A").ColumnWidth = 22
        ws.Columns("B").ColumnWidth = 10
        ws.Columns("C").ColumnWidth = 30
        ws.Columns("D").ColumnWidth = 10
        ws.Columns("E").ColumnWidth = 80
    End If

    Set GetLogSheet = ws
End Function

Private Sub InitLog(yyyymm As Long)
    Dim ws As Worksheet
    Set ws = GetLogSheet()

    Dim nextRow As Long
    nextRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row + 1

    ws.Cells(nextRow, 1).Value = "── NEW RUN ──"
    ws.Cells(nextRow, 2).Value = yyyymm
    ws.Cells(nextRow, 1).Font.Bold = True
    ws.Cells(nextRow, 1).Interior.Color = RGB(180, 180, 180)
    ws.Cells(nextRow, 2).Interior.Color = RGB(180, 180, 180)

    ws.Activate
    Application.ScreenUpdating = True
    DoEvents
End Sub

Private Sub WriteLog(yyyymm As Long, stepName As String, status As String, msg As String)
    Dim ws As Worksheet
    Set ws = GetLogSheet()

    Dim nextRow As Long
    nextRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row + 1

    ws.Cells(nextRow, 1).Value = Format(Now, "DD/MM/YYYY HH:MM:SS")
    ws.Cells(nextRow, 2).Value = yyyymm
    ws.Cells(nextRow, 3).Value = stepName
    ws.Cells(nextRow, 4).Value = status
    ws.Cells(nextRow, 5).Value = msg

    Select Case UCase(status)
        Case "RUNNING": ws.Cells(nextRow, 4).Interior.Color = RGB(255, 255, 180)
        Case "OK":      ws.Cells(nextRow, 4).Interior.Color = RGB(180, 255, 180)
        Case "ERROR":   ws.Cells(nextRow, 4).Interior.Color = RGB(255, 180, 180)
        Case "INFO":    ws.Cells(nextRow, 4).Interior.Color = RGB(180, 220, 255)
    End Select

    Application.ScreenUpdating = True
    DoEvents
End Sub

'─── BUTTON ENTRY POINT ────────────────────────────────────────────────────────

Public Sub RunPipelineManual()
    Dim periodInput As String
    periodInput = InputBox("Enter the period to process (e.g. 202604):", "PF Automation", Format(Now, "YYYYMM"))

    If periodInput = "" Then
        MsgBox "Cancelled.", vbInformation
        Exit Sub
    End If

    If Not IsNumeric(periodInput) Or Len(Trim(periodInput)) <> 6 Then
        MsgBox "Invalid period. Please enter exactly 6 digits e.g. 202604.", vbCritical
        Exit Sub
    End If

    Dim yyyymm As Long
    yyyymm = CLng(periodInput)

    Dim mo As Long
    mo = yyyymm Mod 100
    If mo < 1 Or mo > 12 Then
        MsgBox "Invalid month in " & yyyymm & ". Month must be 01-12.", vbCritical
        Exit Sub
    End If

    Dim srcPath As String
    srcPath = WATCH_FOLDER & periodInput & ".xlsx"
    If Dir(srcPath) = "" Then
        MsgBox "Source file not found:" & vbNewLine & srcPath & vbNewLine & vbNewLine & _
               "Please drop " & periodInput & ".xlsx into:" & vbNewLine & WATCH_FOLDER, vbCritical
        Exit Sub
    End If

    ' Block if QS file already open
    Dim wb As Workbook
    For Each wb In Workbooks
        If InStr(1, LCase(wb.FullName), LCase("QS SQL download")) > 0 Then
            MsgBox "Please close 'QS SQL download.xlsx' before running.", vbCritical
            Exit Sub
        End If
    Next wb

    Dim confirm As Integer
    confirm = MsgBox("Ready to process " & MonthLabel(yyyymm) & " (" & yyyymm & ")." & vbNewLine & vbNewLine & _
                     "Source:  " & srcPath & vbNewLine & _
                     "QS File: " & QS_FILE & vbNewLine & _
                     "Output:  " & OUTPUT_FOLDER & "PF " & MonthLabel(yyyymm) & " Caselist..." & vbNewLine & vbNewLine & _
                     "Draft email will open in Outlook for review." & vbNewLine & vbNewLine & _
                     "Proceed?", vbYesNo + vbQuestion, "PF Automation")

    If confirm = vbNo Then Exit Sub

    Call RunPipeline(srcPath, yyyymm)
End Sub

'─── MAIN PIPELINE ─────────────────────────────────────────────────────────────

Private Sub RunPipeline(srcPath As String, yyyymm As Long)
    Application.DisplayAlerts  = False
    Application.ScreenUpdating = False

    Call InitLog(yyyymm)
    Call WriteLog(yyyymm, "Pipeline", "INFO", "Starting pipeline for " & MonthLabel(yyyymm) & " (" & yyyymm & ")")

    Dim srcWb As Workbook
    Dim srcWs As Worksheet
    Dim qsWb  As Workbook
    Dim qsWs  As Worksheet

    On Error GoTo PipelineError

    '── Step 1: Validate ───────────────────────────────────────────────────────
    Call WriteLog(yyyymm, "Step 1/7 - Validate", "RUNNING", "Opening: " & srcPath)
    Application.ScreenUpdating = False
    Set srcWb = Workbooks.Open(srcPath, ReadOnly:=True)
    Set srcWs = srcWb.Sheets(1)

    Dim errMsg As String
    errMsg = RunChecks(srcWs, yyyymm)
    If errMsg <> "" Then
        Call WriteLog(yyyymm, "Step 1/7 - Validate", "ERROR", errMsg)
        srcWb.Close False
        Application.DisplayAlerts  = True
        Application.ScreenUpdating = True
        MsgBox "Validation failed:" & vbNewLine & errMsg, vbCritical
        Exit Sub
    End If
    Call WriteLog(yyyymm, "Step 1/7 - Validate", "OK", "Validation passed. Period data found.")

    '── Step 2: Open QS file ───────────────────────────────────────────────────
    Call WriteLog(yyyymm, "Step 2/7 - Open QS File", "RUNNING", "Opening: " & QS_FILE)
    Application.ScreenUpdating = False
    Set qsWb = Workbooks.Open(QS_FILE)
    Set qsWs = qsWb.Sheets(QS_DATA_TAB)
    Call WriteLog(yyyymm, "Step 2/7 - Open QS File", "OK", "QS SQL download.xlsx opened.")

    '── Step 3: Delete existing rows ───────────────────────────────────────────
    Call WriteLog(yyyymm, "Step 3/7 - Delete Rows", "RUNNING", "Removing existing rows for " & yyyymm & "...")
    Application.ScreenUpdating = False
    Dim delCount As Long
    delCount = DeleteExistingRows(qsWs, yyyymm)
    Call WriteLog(yyyymm, "Step 3/7 - Delete Rows", "OK", delCount & " rows removed.")

    '── Step 4: Bulk copy filtered rows ────────────────────────────────────────
    Call WriteLog(yyyymm, "Step 4/7 - Copy Data", "RUNNING", "Bulk copying rows where column AA = " & yyyymm & "...")
    Application.ScreenUpdating = False
    Dim copyCount As Long
    copyCount = CopyFilteredRows(srcWs, qsWs, yyyymm)
    srcWb.Close False
    Call WriteLog(yyyymm, "Step 4/7 - Copy Data", "OK", copyCount & " rows pasted into QS data (bulk array method).")

    '── Step 5: Write formulas ─────────────────────────────────────────────────
    Dim lookupCol As String
    lookupCol = GetLookupCol(yyyymm)
    Call WriteLog(yyyymm, "Step 5/7 - Formulas", "RUNNING", "Writing AE/AF formulas. Lookup col = " & lookupCol)
    Application.ScreenUpdating = False
    Call WriteFormulas(qsWs, yyyymm, lookupCol)
    Call WriteLog(yyyymm, "Step 5/7 - Formulas", "OK", "AE/AF formulas written for " & copyCount & " rows.")

    '── Step 6: Refresh pivots ─────────────────────────────────────────────────
    Call WriteLog(yyyymm, "Step 6/7 - Refresh Pivots", "RUNNING", "Refreshing all pivots (waiting 8s)...")
    Application.ScreenUpdating = False
    qsWb.RefreshAll
    Application.Wait Now + TimeValue("00:00:08")
    Call WriteLog(yyyymm, "Step 6/7 - Refresh Pivots", "OK", "Pivots refreshed.")

    '── Step 7: Copy pivots to output ──────────────────────────────────────────
    Dim outPath As String
    outPath = OUTPUT_FOLDER & "PF " & MonthLabel(yyyymm) & _
              " Caselist - EOP Asset Balance and Total Income by Country.xlsx"
    Call WriteLog(yyyymm, "Step 7/7 - Output File", "RUNNING", "Creating output file using PivotTable.TableRange2...")
    Application.ScreenUpdating = False

    Dim niiData() As Variant
    Dim feeData()  As Variant
    Call CopyPivotsToOutput(qsWb, outPath, niiData, feeData)

    qsWb.Save
    qsWb.Close
    Call WriteLog(yyyymm, "Step 7/7 - Output File", "OK", "Output file saved: " & outPath)

    '── MoM analysis ───────────────────────────────────────────────────────────
    Call WriteLog(yyyymm, "MoM Analysis", "RUNNING", "Calculating NII and Fee contributors (>=80%)...")
    Dim niiText As String, feeText As String
    niiText = GetMomContributors(niiData, "NII", yyyymm)
    feeText = GetMomContributors(feeData, "Fee", yyyymm)
    Call WriteLog(yyyymm, "MoM - NII", "INFO", niiText)
    Call WriteLog(yyyymm, "MoM - Fee", "INFO", feeText)
    Call WriteLog(yyyymm, "MoM Analysis", "OK", "Done.")

    '── Draft email ────────────────────────────────────────────────────────────
    Call WriteLog(yyyymm, "Email", "RUNNING", "Opening draft in Outlook...")
    Application.ScreenUpdating = True
    Call SendEmail(outPath, yyyymm, niiText, feeText)
    Call WriteLog(yyyymm, "Email", "OK", "Draft opened. Review and send manually.")
    Call WriteLog(yyyymm, "Pipeline", "OK", "ALL STEPS COMPLETE for " & MonthLabel(yyyymm) & " ✓")

    Application.DisplayAlerts  = True
    Application.ScreenUpdating = True
    MsgBox "Complete for " & MonthLabel(yyyymm) & "!" & vbNewLine & vbNewLine & _
           outPath & vbNewLine & vbNewLine & _
           "Draft email is open — please review and send.", vbInformation, "Done"
    Exit Sub

PipelineError:
    Dim errNum  As Long:   errNum  = Err.Number
    Dim errDesc As String: errDesc = Err.Description
    On Error Resume Next
    If Not srcWb Is Nothing Then srcWb.Close False
    If Not qsWb  Is Nothing Then qsWb.Close False
    On Error GoTo 0
    Application.CutCopyMode = False
    Call WriteLog(yyyymm, "PIPELINE ERROR", "ERROR", "Error " & errNum & ": " & errDesc)
    Application.DisplayAlerts  = True
    Application.ScreenUpdating = True
    MsgBox "Pipeline failed!" & vbNewLine & "Error " & errNum & ": " & errDesc & vbNewLine & vbNewLine & _
           "See Log sheet for details.", vbCritical, "Error"
End Sub

'─── DATA CHECKS ───────────────────────────────────────────────────────────────

Private Function RunChecks(ws As Worksheet, yyyymm As Long) As String
    Dim errors As String
    errors = ""

    If ws.UsedRange.Columns.Count < 27 Then
        errors = errors & "- Source file has fewer than 27 columns." & vbNewLine
        RunChecks = errors
        Exit Function
    End If

    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, COL_AA).End(xlUp).Row

    Dim found As Boolean
    found = False
    Dim i As Long
    For i = 2 To lastRow
        If IsNumeric(ws.Cells(i, COL_AA).Value) Then
            If CLng(ws.Cells(i, COL_AA).Value) = yyyymm Then
                found = True
                Exit For
            End If
        End If
    Next i

    If Not found Then
        errors = errors & "- No rows found where column AA = " & yyyymm & "." & vbNewLine
    End If

    RunChecks = errors
End Function

'─── DELETE EXISTING ROWS ──────────────────────────────────────────────────────

Private Function DeleteExistingRows(ws As Worksheet, yyyymm As Long) As Long
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, COL_AA).End(xlUp).Row
    Dim count As Long
    count = 0
    Dim i As Long
    For i = lastRow To 2 Step -1
        If IsNumeric(ws.Cells(i, COL_AA).Value) Then
            If CLng(ws.Cells(i, COL_AA).Value) = yyyymm Then
                ws.Rows(i).Delete
                count = count + 1
            End If
        End If
    Next i
    DeleteExistingRows = count
End Function

'─── BULK COPY FILTERED ROWS (array method - fast) ─────────────────────────────

Private Function CopyFilteredRows(srcWs As Worksheet, qsWs As Worksheet, yyyymm As Long) As Long
    Dim srcLastRow As Long
    srcLastRow = srcWs.Cells(srcWs.Rows.Count, COL_AA).End(xlUp).Row

    If srcLastRow < 2 Then
        CopyFilteredRows = 0
        Exit Function
    End If

    ' Read entire source into memory at once
    Dim srcData() As Variant
    srcData = srcWs.Range(srcWs.Cells(2, 1), srcWs.Cells(srcLastRow, 30)).Value

    ' Count matching rows
    Dim count As Long
    count = 0
    Dim i As Long
    For i = 1 To UBound(srcData, 1)
        If IsNumeric(srcData(i, COL_AA)) Then
            If CLng(srcData(i, COL_AA)) = yyyymm Then count = count + 1
        End If
    Next i

    If count = 0 Then
        CopyFilteredRows = 0
        Exit Function
    End If

    ' Build filtered array
    Dim outData() As Variant
    ReDim outData(1 To count, 1 To 30)
    Dim destIdx As Long
    destIdx = 0
    Dim j As Long
    For i = 1 To UBound(srcData, 1)
        If IsNumeric(srcData(i, COL_AA)) Then
            If CLng(srcData(i, COL_AA)) = yyyymm Then
                destIdx = destIdx + 1
                For j = 1 To 30
                    outData(destIdx, j) = srcData(i, j)
                Next j
            End If
        End If
    Next i

    ' Write all rows in one shot
    Dim destRow As Long
    destRow = qsWs.Cells(qsWs.Rows.Count, 1).End(xlUp).Row + 1
    qsWs.Cells(destRow, 1).Resize(count, 30).Value = outData

    CopyFilteredRows = count
End Function

'─── WRITE AE / AF FORMULAS ────────────────────────────────────────────────────

Private Sub WriteFormulas(ws As Worksheet, yyyymm As Long, lookupCol As String)
    Dim lastRow As Long
    lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row

    Dim i As Long
    For i = 2 To lastRow
        If IsNumeric(ws.Cells(i, COL_AA).Value) Then
            If CLng(ws.Cells(i, COL_AA).Value) = yyyymm Then
                ws.Cells(i, COL_AE).Formula = "=F" & i & "&E" & i & "&N" & i
                ws.Cells(i, COL_AF).Formula = _
                    "=IF(ISNUMBER(SEARCH(""card"",Q" & i & ")),""Card""," & _
                    "IFERROR(VLOOKUP(AE" & i & ",'" & CHECK_DUP_TAB & "'!" & _
                    lookupCol & ":" & lookupCol & ",1,0),""*No""))"
            End If
        End If
    Next i
End Sub

'─── COPY FULL PIVOTS TO OUTPUT (uses PivotTable.TableRange2) ──────────────────

Private Sub CopyPivotsToOutput(qsWb As Workbook, outPath As String, _
                                niiData() As Variant, feeData() As Variant)
    Dim pivotWs As Worksheet
    Set pivotWs = qsWb.Sheets(PIVOT_TAB)

    ' Output workbook
    Dim outWb As Workbook
    Set outWb = Workbooks.Add
    Do While outWb.Sheets.Count > 1
        outWb.Sheets(outWb.Sheets.Count).Delete
    Loop

    ' Pivot start cells and target tab names
    Dim pivotCells(1 To 5) As String
    Dim tabNames(1 To 5)   As String
    pivotCells(1) = "A1":  tabNames(1) = "EOP Asset Balance"
    pivotCells(2) = "S1":  tabNames(2) = "Total Income"
    pivotCells(3) = "AP1": tabNames(3) = "NII"
    pivotCells(4) = "BH1": tabNames(4) = "Fee"
    pivotCells(5) = "BZ1": tabNames(5) = "Other income"

    Dim idx As Long
    For idx = 1 To 5
        Dim startCell As Range
        Set startCell = pivotWs.Range(pivotCells(idx))

        ' Find the PivotTable that starts at (or contains) this cell
        Dim pvRange As Range
        Dim pt      As PivotTable
        Dim found   As Boolean
        found = False

        For Each pt In pivotWs.PivotTables
            ' Check if pivot starts at this cell
            If pt.TableRange2.Cells(1, 1).Address = startCell.Address Then
                Set pvRange = pt.TableRange2
                found = True
                Exit For
            End If
        Next pt

        ' Fallback: find pivot that contains the cell
        If Not found Then
            For Each pt In pivotWs.PivotTables
                If Not Intersect(pt.TableRange2, startCell) Is Nothing Then
                    Set pvRange = pt.TableRange2
                    found = True
                    Exit For
                End If
            Next pt
        End If

        ' Final fallback: CurrentRegion
        If Not found Then
            Set pvRange = startCell.CurrentRegion
        End If

        ' Store values for MoM analysis
        Dim pvData() As Variant
        pvData = pvRange.Value
        If idx = 3 Then niiData = pvData
        If idx = 4 Then feeData = pvData

        ' Create output sheet
        Dim outWs As Worksheet
        If idx = 1 Then
            Set outWs = outWb.Sheets(1)
            outWs.Name = tabNames(idx)
        Else
            Set outWs = outWb.Sheets.Add(After:=outWb.Sheets(outWb.Sheets.Count))
            outWs.Name = tabNames(idx)
        End If

        ' Copy values and formats into output sheet
        pvRange.Copy
        outWs.Range("A1").PasteSpecial xlPasteValues
        outWs.Range("A1").PasteSpecial xlPasteFormats
        Application.CutCopyMode = False
    Next idx

    outWb.SaveAs outPath, xlOpenXMLWorkbook
    outWb.Close
End Sub

'─── MOM ANALYSIS ──────────────────────────────────────────────────────────────

Private Function GetMomContributors(pvData() As Variant, label As String, yyyymm As Long) As String
    On Error GoTo ErrHandler

    If Not IsArray(pvData) Then
        GetMomContributors = label & ": no data available."
        Exit Function
    End If

    Dim prevMm As Long
    prevMm = PrevYYYYMM(yyyymm)

    Dim currCol As Long, prevCol As Long
    currCol = 0: prevCol = 0
    Dim c As Long
    For c = 1 To UBound(pvData, 2)
        If IsNumeric(pvData(1, c)) Then
            If CLng(pvData(1, c)) = yyyymm Then currCol = c
            If CLng(pvData(1, c)) = prevMm  Then prevCol = c
        End If
    Next c

    If currCol = 0 Or prevCol = 0 Then
        GetMomContributors = label & ": could not find columns for " & yyyymm & " or " & prevMm
        Exit Function
    End If

    Dim names()  As String
    Dim moves()  As Double
    Dim count    As Long
    count = 0

    Dim r As Long
    For r = 2 To UBound(pvData, 1)
        Dim rowLabel As String
        rowLabel = ""
        If Not IsEmpty(pvData(r, 1)) Then rowLabel = Trim(CStr(pvData(r, 1)))
        If rowLabel = "" Or LCase(rowLabel) = "grand total" Then GoTo NextRow

        Dim currVal As Double, prevVal As Double
        currVal = 0: prevVal = 0
        If IsNumeric(pvData(r, currCol)) Then currVal = CDbl(pvData(r, currCol))
        If IsNumeric(pvData(r, prevCol)) Then prevVal = CDbl(pvData(r, prevCol))

        Dim movement As Double
        movement = currVal - prevVal
        If movement <> 0 Then
            count = count + 1
            ReDim Preserve names(1 To count)
            ReDim Preserve moves(1 To count)
            names(count) = rowLabel
            moves(count) = movement
        End If
NextRow:
    Next r

    If count = 0 Then
        GetMomContributors = label & ": no MoM movement detected."
        Exit Function
    End If

    Dim j As Long
    Dim tempN As String, tempM As Double
    For r = 1 To count - 1
        For j = 1 To count - r
            If Abs(moves(j)) < Abs(moves(j + 1)) Then
                tempN = names(j): tempM = moves(j)
                names(j) = names(j + 1): moves(j) = moves(j + 1)
                names(j + 1) = tempN: moves(j + 1) = tempM
            End If
        Next j
    Next r

    Dim totalMove As Double, absTotal As Double
    totalMove = 0: absTotal = 0
    For r = 1 To count
        totalMove = totalMove + moves(r)
        absTotal  = absTotal  + Abs(moves(r))
    Next r

    Dim result As String
    result = label & " MoM (" & MonthLabel(prevMm) & " to " & MonthLabel(yyyymm) & "): " & _
             Format(Abs(totalMove), "#,##0") & IIf(totalMove >= 0, " up", " down") & vbNewLine & _
             "Key contributors (>=80%):" & vbNewLine

    Dim cumulative As Double
    cumulative = 0
    For r = 1 To count
        Dim pct As Double
        pct = IIf(totalMove <> 0, moves(r) / totalMove * 100, 0)
        result = result & "  " & IIf(moves(r) >= 0, Chr(9650), Chr(9660)) & " " & _
                 names(r) & ": " & Format(moves(r), "+#,##0;-#,##0") & _
                 " (" & Format(pct, "+0.0;-0.0") & "%)" & vbNewLine
        cumulative = cumulative + Abs(moves(r))
        If absTotal > 0 And cumulative / absTotal >= 0.8 Then Exit For
    Next r

    GetMomContributors = result
    Exit Function
ErrHandler:
    GetMomContributors = label & ": error - " & Err.Description
End Function

'─── SEND EMAIL ────────────────────────────────────────────────────────────────

Private Sub SendEmail(attachPath As String, yyyymm As Long, niiText As String, feeText As String)
    Dim olApp  As Object
    Dim olMail As Object
    Set olApp  = CreateObject("Outlook.Application")
    Set olMail = olApp.CreateItem(0)

    Dim body As String
    body = "Hi Steve," & vbNewLine & vbNewLine & _
           "Please find attached the updated PF Caselist - " & _
           "EOP Asset Balance and Total Income by Country." & _
           vbNewLine & vbNewLine & _
           niiText & vbNewLine & _
           feeText & vbNewLine & _
           "Best regards," & vbNewLine & EMAIL_NAME

    With olMail
        .To      = EMAIL_TO
        .CC      = EMAIL_CC
        .Subject = "PF " & MonthLabel(yyyymm) & _
                   " Caselist - EOP Asset Balance and Total Income by Country"
        .Body    = body
        .Attachments.Add attachPath
        .Display
    End With
End Sub

'─── HELPERS ───────────────────────────────────────────────────────────────────

Private Function GetLookupCol(yyyymm As Long) As String
    Dim diff   As Long
    Dim colIdx As Long
    diff   = ((yyyymm \ 100) - (BASE_MONTH \ 100)) * 12 + _
             ((yyyymm Mod 100) - (BASE_MONTH Mod 100))
    colIdx = BASE_COL_IDX + diff
    If colIdx <= 26 Then
        GetLookupCol = Chr(64 + colIdx)
    Else
        GetLookupCol = Chr(64 + (colIdx - 1) \ 26) & Chr(65 + (colIdx - 1) Mod 26)
    End If
End Function

Private Function PrevYYYYMM(yyyymm As Long) As Long
    Dim yr As Long, mo As Long
    yr = yyyymm \ 100
    mo = yyyymm Mod 100
    If mo = 1 Then
        PrevYYYYMM = (yr - 1) * 100 + 12
    Else
        PrevYYYYMM = yr * 100 + (mo - 1)
    End If
End Function

Private Function MonthLabel(yyyymm As Long) As String
    Dim mNames(1 To 12) As String
    mNames(1) = "Jan": mNames(2) = "Feb": mNames(3)  = "Mar"
    mNames(4) = "Apr": mNames(5) = "May": mNames(6)  = "Jun"
    mNames(7) = "Jul": mNames(8) = "Aug": mNames(9)  = "Sep"
    mNames(10) = "Oct": mNames(11) = "Nov": mNames(12) = "Dec"
    MonthLabel = mNames(yyyymm Mod 100) & " " & Right(CStr(yyyymm \ 100), 2)
End Function
