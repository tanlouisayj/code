'=============================================================================
' PF FINANCIALS AUTOMATION - PART 2 MODULE  (v8)
' Paste into a NEW Module in VBA Editor (Insert > Module → Module2)
' Works alongside PF_Automation_v7 (Module1) — no changes to Module1 needed
'=============================================================================
' CHANGES IN v8 (vs v7):
'   - P2_UpdateDCLinks (Step 3):
'       * Completely rewrote row-29/30 write loop into two separate blocks.
'       * Row 30 now has its own template scan (Strategy A): scans ALL cols
'         1..currCol in row 30 for a valid "='" external-link formula, derives
'         its own column offset and month label — independent of row 29's template.
'       * Strategy B fallback (no valid row-30 template found): derives row 30's
'         formula from the newly-built row-29 formula by replacing only the
'         external-file row number (e.g. $31 → $32), reading the target row
'         number from any existing working row-30 formula in earlier columns.
'       * Root cause fixed: old code read Cells(30, tmplCol).Formula which
'         returned garbage on a broken/not-yet-open link, then fell back to
'         tmplF (row-29 template), so row 30 got the same $row$ reference as
'         row 29 — both ended up pointing at the same source row.
' CHANGES IN v7 (vs v6):
'   - P2_UpdateDCLinks (Step 3):
'       * Rewrote column-advance logic: instead of text-substituting the prev
'         month label in the formula string, now scans row 8 on P&L_PF for the
'         current reporting month header to find currCol, derives prevCol = currCol-1,
'         copies prevCol formula to currCol using FillRight-equivalent column-letter
'         increment. Handles Z→AA and any two-letter column transitions correctly.
'       * P2_IncrementColInFormula helper added: finds every column letter
'         reference in an external-link formula and advances it by +1, so
'         [Book.xlsm]Sheet!$Z$5 becomes [Book.xlsm]Sheet!$AA$5 etc.
'       * Month-label text in the formula is also updated (Mar 26 → Apr 26).
'   - P2_BuildCommentary:
'       * bracketNIM confirmed col N (14) row 14 — YTD NIM for compressed NIM bracket.
'       * currNIM col I (9) row 14 — current month NIM for MoM sentence (unchanged).
' CHANGES IN v6 (vs v5):
'   - P2_CreateSentFile:
'       * Step 7 (new): ClearComments on ALL sheets before SaveAs — ensures no
'         cell notes are present in the sent .xlsx file
'   - P2_BuildCommentary:
'       * Added bracketNIM = P2V(dashWs, 14, CN)  [col N row 14]
'         Used ONLY in the NII sentence "compressed NIM (x% vs y% budget)"
'       * currNIM (col I row 14) is still used for the NIM MoM sentence
'         "NIM decreased xxbps MoM from x% in Mar to y% in Apr"
' CHANGES IN v5 (vs v4):
'   - P2_BuildCommentary:
'       * currNIM reads row 14 col I (9) — was col N (14) in v4
'       * Extra vbNewLine before X-selling section removed
'   - P2_GetFeeContributors:
'       * Customer name reads col 1 (A) first, falls back to col 2 (B)
' CHANGES IN v4 (vs v3):
'   - P2_BuildCommentary: currEOP col N; Non-NII = I24-H24; totals passed to fees
'   - P2_GetFeeContributors: 80% cumulative coverage replaces fixed top-3 cap
'=============================================================================

Const T_WORK_PIVOTS  As String = "Working + Working PF pivots"
Const T_NORM_NIM     As String = "Normalised NIM"

Const T_PL_PF        As String = "P&L_PF"

'─── SETTINGS READERS ──────────────────────────────────────────────────────────

Private Function P2S(cell As String) As String
    P2S = Trim(CStr(ThisWorkbook.Sheets("Sheet1").Range(cell).Value))
End Function

Private Function P2_Folder() As String
    Dim p As String: p = P2S(P2_FOLDER_CELL)
    If Right(p, 1) <> "\" Then p = p & "\"
    P2_Folder = p
End Function

Private Function P2_QSFile() As String:     P2_QSFile    = P2S(P2_QS_FILE_CELL):   End Function
Private Function P2_EmailTo() As String:    P2_EmailTo   = P2S(P2_EMAIL_TO_CELL):  End Function
Private Function P2_EmailCC() As String:    P2_EmailCC   = P2S(P2_EMAIL_CC_CELL):  End Function
Private Function P2_Greeting() As String:   P2_Greeting  = P2S(P2_GREETING_CELL):  End Function
Private Function P2_YourName() As String:   P2_YourName  = P2S(P2_YOUR_NAME_CELL): End Function

Private Function P2_NimStart() As Long
    Dim s As String: s = P2S(P2_NIM_RANGE_CELL)
    Dim p() As String: p = Split(s, ":")
    If UBound(p) >= 0 And IsNumeric(p(0)) Then P2_NimStart = CLng(p(0)) Else P2_NimStart = 298
End Function

Private Function P2_NimEnd() As Long
    Dim s As String: s = P2S(P2_NIM_RANGE_CELL)
    Dim p() As String: p = Split(s, ":")
    If UBound(p) >= 1 And IsNumeric(p(1)) Then P2_NimEnd = CLng(p(1)) Else P2_NimEnd = 320
End Function

Private Function P2_DataStart() As Long
    Dim s As String: s = P2S(P2_NIM_DATA_CELL)
    Dim p() As String: p = Split(s, ":")
    If UBound(p) >= 0 And IsNumeric(p(0)) Then P2_DataStart = CLng(p(0)) Else P2_DataStart = 10
End Function

Private Function P2_DataEnd() As Long
    Dim s As String: s = P2S(P2_NIM_DATA_CELL)
    Dim p() As String: p = Split(s, ":")
    If UBound(p) >= 1 And IsNumeric(p(1)) Then P2_DataEnd = CLng(p(1)) Else P2_DataEnd = 291
End Function

'─── BUTTON ENTRY POINT ────────────────────────────────────────────────────────

Public Sub RunPFFinancialsManual()
    If P2_Folder() = "" Or P2_QSFile() = "" Then
        MsgBox "Fill in Part 2 settings on Sheet1 first (rows 36-43).", vbCritical: Exit Sub
    End If

    Dim fso As Object: Set fso = CreateObject("Scripting.FileSystemObject")
    Dim folder As Object
    On Error Resume Next: Set folder = fso.GetFolder(P2_Folder()): On Error GoTo 0

    If folder Is Nothing Then
        MsgBox "Cannot access folder:" & vbNewLine & P2_Folder(), vbCritical: Exit Sub
    End If

    Dim workingPath As String: workingPath = ""
    Dim file As Object
    For Each file In folder.Files
        If InStr(1, LCase(file.Name), "project_finance") > 0 And _
           InStr(1, LCase(file.Name), "(working)") > 0 Then
            workingPath = file.Path: Exit For
        End If
    Next file

    If workingPath = "" Then
        MsgBox "No 'Project_Finance * (working).*' file found in:" & vbNewLine & P2_Folder(), vbCritical
        Exit Sub
    End If

    Dim oldPeriod As Long: oldPeriod = P2_ExtractPeriod(fso.GetFileName(workingPath))
    Dim newPeriod As Long: newPeriod = P2_NextYYYYMM(oldPeriod)

    If newPeriod = 0 Then
        MsgBox "Could not determine period from filename: " & fso.GetFileName(workingPath), vbCritical
        Exit Sub
    End If

    Dim wb As Workbook
    For Each wb In Workbooks
        If InStr(1, LCase(wb.FullName), LCase("QS SQL download")) > 0 Then
            MsgBox "Please close 'QS SQL download.xlsx' before running.", vbCritical: Exit Sub
        End If
    Next wb

    If MsgBox("Current working file:  " & fso.GetFileName(workingPath) & vbNewLine & vbNewLine & _
              "Will rename to:  Project_Finance " & P2_MonthLabel(newPeriod) & " (working).xlsm" & vbNewLine & _
              "Generating PF Financials for:  " & P2_MonthLabel(newPeriod) & vbNewLine & vbNewLine & _
              "To:  " & P2_EmailTo() & vbNewLine & _
              "CC:  " & P2_EmailCC() & vbNewLine & vbNewLine & _
              "Proceed?", vbYesNo + vbQuestion, "PF Financials Part 2") = vbNo Then Exit Sub

    Call RunPFFinancials(workingPath, newPeriod)
End Sub

'─── MAIN PIPELINE ─────────────────────────────────────────────────────────────

Private Sub RunPFFinancials(workingPath As String, yyyymm As Long)
    Application.DisplayAlerts = False
    Application.ScreenUpdating = False

    Dim pfWb  As Workbook
    Dim qsWb  As Workbook

    Dim logWs As Worksheet
    On Error Resume Next: Set logWs = ThisWorkbook.Sheets("Log"): On Error GoTo 0
    If logWs Is Nothing Then
        Set logWs = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        logWs.Name = "Log"
        logWs.Cells(1, 1).Value = "Timestamp":  logWs.Cells(1, 2).Value = "Source"
        logWs.Cells(1, 3).Value = "Step":       logWs.Cells(1, 4).Value = "Status"
        logWs.Cells(1, 5).Value = "Message"
        logWs.Rows(1).Font.Bold = True
    End If
    Dim sepRow As Long: sepRow = logWs.Cells(logWs.Rows.Count, 1).End(xlUp).Row + 1
    logWs.Cells(sepRow, 1).Value = "── PART2 RUN ──"
    logWs.Cells(sepRow, 2).Value = yyyymm
    logWs.Cells(sepRow, 1).Font.Bold = True
    logWs.Cells(sepRow, 1).Interior.Color = RGB(180, 220, 180)
    logWs.Cells(sepRow, 2).Interior.Color = RGB(180, 220, 180)
    logWs.Activate

    On Error GoTo PipelineError2

    '── Step 1: Rename working file ────────────────────────────────────────────
    P2Log "Step 1/9 - Rename File", "RUNNING", "Renaming to Project_Finance " & P2_MonthLabel(yyyymm) & " (working)..."
    Dim fso As Object: Set fso = CreateObject("Scripting.FileSystemObject")
    Dim newFileName As String
    newFileName = "Project_Finance " & P2_MonthLabel(yyyymm) & " (working).xlsm"
    Dim newPath As String: newPath = P2_Folder() & newFileName

    If Dir(newPath) = "" Then fso.MoveFile workingPath, newPath
    Application.ScreenUpdating = False
    Set pfWb = Workbooks.Open(newPath)
    P2Log "Step 1/9 - Rename File", "OK", "Opened: " & newFileName

    '── Step 2: Open QS SQL download → refresh pivots + Normalised NIM ─────────
    P2Log "Step 2/9 - Refresh QS Pivots", "RUNNING", "Opening: " & P2_QSFile()
    Application.ScreenUpdating = False
    Set qsWb = Workbooks.Open(P2_QSFile())

    P2_RefreshTabPivots qsWb, T_WORK_PIVOTS
    P2_RefreshNIMTab qsWb, yyyymm

    Dim normNIM As Double: normNIM = P2_GetNormNIM(qsWb, yyyymm)
    P2Log "Step 2/9 - Refresh QS Pivots", "OK", _
          "QS pivots refreshed. Normalised NIM = " & Format(normNIM * 100, "0.00") & "%"

    qsWb.Save
    qsWb.Close
    Set qsWb = Nothing

    '── Step 3: Update Direct Cost links rows 29-30 ────────────────────────────
    P2Log "Step 3/9 - Direct Cost Links", "RUNNING", "Updating P&L_PF rows 29-30..."
    Application.ScreenUpdating = False
    Dim plWs As Worksheet: Set plWs = pfWb.Sheets(T_PL_PF)
    P2_UpdateDCLinks plWs, yyyymm
    P2Log "Step 3/9 - Direct Cost Links", "OK", "Done."

    '── Step 4: Refresh all workbook links ─────────────────────────────────────
    P2Log "Step 4/9 - Refresh Links", "RUNNING", "Refreshing workbook links..."
    Application.ScreenUpdating = False
    On Error Resume Next
    pfWb.UpdateLink Name:=pfWb.LinkSources, Type:=xlExcelLinks
    On Error GoTo PipelineError2
    Application.Wait Now + TimeValue("00:00:04")
    Application.Calculate
    P2Log "Step 4/9 - Refresh Links", "OK", "Links refreshed."

    '── Step 5: Refresh reporting tab pivots ───────────────────────────────────
    P2Log "Step 5/9 - Refresh Report Pivots", "RUNNING", "Refreshing PF Fees, Income by Country, etc..."
    Application.ScreenUpdating = False
    P2_RefreshTabPivots pfWb, T_PF_FEES
    P2_RefreshTabPivots pfWb, T_INC_CNTRY
    P2_RefreshTabPivots pfWb, T_XSELL
    P2_RefreshTabPivots pfWb, T_CUST_LIST
    Application.Calculate
    Application.Wait Now + TimeValue("00:00:03")
    P2Log "Step 5/9 - Refresh Report Pivots", "OK", "Done."

    '── Step 6: Build and update commentary text box ───────────────────────────
    P2Log "Step 6/9 - Update Commentary", "RUNNING", "Building commentary from P&L cells..."
    Application.ScreenUpdating = False
    Dim commentary As String
    commentary = P2_BuildCommentary(plWs, yyyymm, normNIM)
    Dim dashWs As Worksheet: Set dashWs = pfWb.Sheets(T_PL_DASH)
    P2_UpdateTextBox dashWs, commentary
    P2Log "Step 6/9 - Update Commentary", "OK", "Commentary updated. Review [UPDATE] placeholders."

    '── Step 7: Save working file ──────────────────────────────────────────────
    P2Log "Step 7/9 - Save Working", "RUNNING", "Saving " & newFileName & "..."
    Application.ScreenUpdating = False
    pfWb.Save
    P2Log "Step 7/9 - Save Working", "OK", "Saved."

    '── Step 8: Create sent file ───────────────────────────────────────────────
    Dim sentFileName As String
    sentFileName = "Project_Finance " & P2_MonthLabel(yyyymm) & " (sent).xlsx"
    Dim sentPath As String: sentPath = P2_Folder() & sentFileName
    P2Log "Step 8/9 - Create Sent File", "RUNNING", "Creating: " & sentFileName
    Application.ScreenUpdating = False
    P2_CreateSentFile pfWb, yyyymm, sentPath
    P2Log "Step 8/9 - Create Sent File", "OK", "Sent file created."

    '── Step 9: Screenshot + draft email ───────────────────────────────────────
    P2Log "Step 9/9 - Email Draft", "RUNNING", "Capturing dashboard and drafting email..."
    Application.ScreenUpdating = True

    Dim imgPath As String
    imgPath = Environ("TEMP") & "\PF_Dashboard_" & yyyymm & ".png"
    Dim imgOK As Boolean: imgOK = False

    Dim sentWb As Workbook: Set sentWb = Workbooks.Open(sentPath)
    On Error Resume Next
    Dim sentDash As Worksheet: Set sentDash = sentWb.Sheets(T_PL_DASH)
    On Error GoTo PipelineError2
    If Not sentDash Is Nothing Then
        imgOK = P2_CaptureImage(sentDash, "B6:X25", imgPath)
    End If
    sentWb.Close False

    P2_SendEmail sentPath, imgPath, imgOK, commentary, yyyymm
    P2Log "Step 9/9 - Email Draft", "OK", "Draft email opened. Review [UPDATE] items and send."
    P2Log "ALL COMPLETE", "OK", "PF Financials for " & P2_MonthLabel(yyyymm) & " done ✓"

    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    MsgBox "Part 2 complete for " & P2_MonthLabel(yyyymm) & "!" & vbNewLine & vbNewLine & _
           "Working: " & newPath & vbNewLine & _
           "Sent:    " & sentPath & vbNewLine & vbNewLine & _
           "Review [UPDATE] placeholders in the draft email before sending.", _
           vbInformation, "Part 2 Done"
    Exit Sub

PipelineError2:
    Dim en As Long: en = Err.Number
    Dim ed As String: ed = Err.Description
    On Error Resume Next
    If Not pfWb Is Nothing Then pfWb.Close False
    If Not qsWb Is Nothing Then qsWb.Close False
    Err.Clear
    On Error GoTo 0
    P2Log "PIPELINE ERROR", "ERROR", "Error " & en & ": " & ed
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    MsgBox "Error " & en & ": " & ed & vbNewLine & vbNewLine & _
           "Check Log sheet for details.", vbCritical, "Part 2 Error"
End Sub

'─── REFRESH TAB PIVOTS ────────────────────────────────────────────────────────

Private Sub P2_RefreshTabPivots(wb As Workbook, tabName As String)
    On Error Resume Next
    Dim ws As Worksheet: Set ws = wb.Sheets(tabName)
    On Error GoTo 0
    If ws Is Nothing Then Exit Sub
    Dim pt As PivotTable
    For Each pt In ws.PivotTables
        pt.RefreshTable
    Next pt
    ws.Calculate
End Sub

'─── NORMALISED NIM ────────────────────────────────────────────────────────────

Private Sub P2_RefreshNIMTab(qsWb As Workbook, yyyymm As Long)
    On Error Resume Next
    Dim ws As Worksheet: Set ws = qsWb.Sheets(T_NORM_NIM)
    On Error GoTo 0
    If ws Is Nothing Then Exit Sub

    Dim pt As PivotTable
    For Each pt In ws.PivotTables
        pt.RefreshTable
    Next pt
    Application.Calculate
    Application.Wait Now + TimeValue("00:00:02")

    P2_UpdateNIMFormulas ws, yyyymm
    P2_WriteNIMPercentFormula ws, yyyymm
    ws.Calculate
End Sub

Private Sub P2_UpdateNIMFormulas(ws As Worksheet, yyyymm As Long)
    Dim mo As Long: mo = yyyymm Mod 100
    Dim yr As Long: yr = yyyymm \ 100
    Dim isQEnd As Boolean: isQEnd = (mo = 3 Or mo = 6 Or mo = 9 Or mo = 12)
    Dim m1 As Long, m2 As Long, m3 As Long, currMM As Long
    currMM = yyyymm

    If isQEnd Then
        Dim qStart As Long: qStart = mo - 2
        m1 = yr * 100 + qStart
        m2 = yr * 100 + qStart + 1
        m3 = yyyymm
    Else
        Select Case mo
            Case 1, 2:   m3 = (yr - 1) * 100 + 12
            Case 4, 5:   m3 = yr * 100 + 3
            Case 7, 8:   m3 = yr * 100 + 6
            Case 10, 11: m3 = yr * 100 + 9
        End Select
        m2 = P2_PrevYYYYMM(m3)
        m1 = P2_PrevYYYYMM(m2)
    End If

    Dim col1 As Long: col1 = P2_FindNIMCol(ws, m1, "NII")
    Dim col2 As Long: col2 = P2_FindNIMCol(ws, m2, "NII")
    Dim col3 As Long: col3 = P2_FindNIMCol(ws, m3, "NII")
    Dim colC As Long: colC = P2_FindNIMCol(ws, currMM, "NII")

    If col1 = 0 Or col2 = 0 Or col3 = 0 Or colC = 0 Then
        P2Log "Normalised NIM", "INFO", "Could not find all NII columns — " & m1 & "," & m2 & "," & m3 & "," & currMM & " — skipping formula update."
        Exit Sub
    End If

    Dim l1 As String: l1 = P2_ColLetter(col1)
    Dim l2 As String: l2 = P2_ColLetter(col2)
    Dim l3 As String: l3 = P2_ColLetter(col3)
    Dim lC As String: lC = P2_ColLetter(colC)
    Dim ds As Long: ds = P2_DataStart()
    Dim de As Long: de = P2_DataEnd()

    Dim r As Long
    For r = P2_NimStart() To P2_NimEnd()
        If Trim(CStr(ws.Cells(r, 2).Value)) <> "" Then
            Dim f As String
            f = "=(SUMIF($B$" & ds & ":$B$" & de & ",$B" & r & ",$" & l1 & "$" & ds & ":$" & l1 & "$" & de & ")" & _
                "+SUMIF($B$" & ds & ":$B$" & de & ",$B" & r & ",$" & l2 & "$" & ds & ":$" & l2 & "$" & de & ")" & _
                "+SUMIF($B$" & ds & ":$B$" & de & ",$B" & r & ",$" & l3 & "$" & ds & ":$" & l3 & "$" & de & "))" & _
                "/3-SUMIF($B$" & ds & ":$B$" & de & ",$B" & r & ",$" & lC & "$" & ds & ":$" & lC & "$" & de & ")"
            ws.Cells(r, colC).Formula = f
        End If
    Next r

    P2Log "Normalised NIM", "INFO", "SUMIF formulas updated. Months: " & _
          P2_MonthLabel(m1) & "+" & P2_MonthLabel(m2) & "+" & P2_MonthLabel(m3) & _
          " cols " & l1 & "+" & l2 & "+" & l3 & " / 3 minus col " & lC
End Sub

Private Sub P2_WriteNIMPercentFormula(ws As Worksheet, yyyymm As Long)
    Dim colNII   As Long: colNII   = P2_FindNIMCol(ws, yyyymm, "NII")
    Dim colAsset As Long: colAsset = colNII + 1

    If colNII = 0 Then
        P2Log "Normalised NIM %", "INFO", "Could not find NII column for " & yyyymm & " — NIM % formula not updated."
        Exit Sub
    End If

    Dim grandTotalRow As Long: grandTotalRow = 0
    Dim lastRow As Long: lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    Dim r As Long
    For r = 1 To lastRow
        If InStr(1, LCase(CStr(ws.Cells(r, 1).Value)), "grand total") > 0 Or _
           InStr(1, LCase(CStr(ws.Cells(r, 2).Value)), "grand total") > 0 Then
            grandTotalRow = r: Exit For
        End If
    Next r

    If grandTotalRow = 0 Then
        P2Log "Normalised NIM %", "INFO", "Grand Total row not found — NIM % formula not updated."
        Exit Sub
    End If

    Dim daysInMo As Long: daysInMo = P2_DaysInMonth(yyyymm)

    Dim nimPctRow As Long: nimPctRow = 0
    Dim nimPctCol As Long: nimPctCol = colNII

    For r = 345 To 360
        If ws.Cells(r, nimPctCol).Interior.Color = RGB(255, 255, 0) Or _
           ws.Cells(r, nimPctCol).Interior.ColorIndex = 6 Or _
           ws.Cells(r, nimPctCol).Interior.ColorIndex = 36 Then
            nimPctRow = r: Exit For
        End If
    Next r

    If nimPctRow = 0 Then
        For r = 345 To 360
            If ws.Cells(r, nimPctCol).HasFormula Then
                nimPctRow = r: Exit For
            End If
        Next r
    End If

    If nimPctRow = 0 Then
        Dim v As Variant
        For r = 345 To 360
            v = ws.Cells(r, nimPctCol).Value
            If IsNumeric(v) Then
                If CDbl(v) > 0.003 And CDbl(v) < 0.03 Then
                    nimPctRow = r: Exit For
                End If
            End If
        Next r
    End If

    If nimPctRow = 0 Then
        nimPctRow = 350
        P2Log "Normalised NIM %", "INFO", "NIM % cell not found by colour/formula — using confirmed row 350."
    End If

    Dim lNII  As String: lNII  = P2_ColLetter(colNII)
    Dim lAss  As String: lAss  = P2_ColLetter(colAsset)
    Dim nimSt As Long:   nimSt = P2_NimStart()
    Dim nimEn As Long:   nimEn = P2_NimEnd()

    Dim nimFormula As String
    nimFormula = "=(SUM(" & lNII & nimSt & ":" & lNII & nimEn & "," & lNII & grandTotalRow & ")" & _
                 "/" & daysInMo & "*365)" & _
                 "/" & lAss & grandTotalRow

    On Error Resume Next
    ws.Cells(nimPctRow, nimPctCol).Formula = nimFormula
    Dim writeErr As Long: writeErr = Err.Number
    On Error GoTo 0

    If writeErr <> 0 Then
        P2Log "Normalised NIM %", "INFO", "Could not write NIM % formula at row " & nimPctRow & _
              " col " & lNII & " (cell may be protected or merged). Formula was: " & nimFormula
    Else
        On Error Resume Next
        ws.Cells(nimPctRow, nimPctCol).NumberFormat = "0.00%"
        On Error GoTo 0
        P2Log "Normalised NIM %", "OK", "NIM % formula written at row " & nimPctRow & _
              " col " & lNII & ". Grand Total row " & grandTotalRow & ". Days = " & daysInMo & "."
    End If
End Sub

Private Function P2_DaysInMonth(yyyymm As Long) As Long
    Dim yr As Long: yr = yyyymm \ 100
    Dim mo As Long: mo = yyyymm Mod 100
    Select Case mo
        Case 1, 3, 5, 7, 8, 10, 12: P2_DaysInMonth = 31
        Case 4, 6, 9, 11:            P2_DaysInMonth = 30
        Case 2
            If (yr Mod 4 = 0 And yr Mod 100 <> 0) Or yr Mod 400 = 0 Then
                P2_DaysInMonth = 29
            Else
                P2_DaysInMonth = 28
            End If
    End Select
End Function

Private Function P2_FindNIMCol(ws As Worksheet, yyyymm As Long, valueLabel As String) As Long
    Dim lastCol As Long: lastCol = ws.Cells(9, ws.Columns.Count).End(xlToLeft).Column
    Dim c As Long
    Dim yyyymmStr As String: yyyymmStr = CStr(yyyymm)
    Dim mmCol As Long: mmCol = 0

    For c = 1 To lastCol
        Dim cellVal As Variant: cellVal = ws.Cells(9, c).Value
        If IsNumeric(cellVal) Then
            If CLng(cellVal) = yyyymm Then mmCol = c: Exit For
        ElseIf CStr(cellVal) = yyyymmStr Then
            mmCol = c: Exit For
        End If
    Next c

    If mmCol = 0 Then P2_FindNIMCol = 0: Exit Function

    Dim endC As Long: endC = Application.Min(mmCol + 10, lastCol)
    For c = mmCol To endC
        If InStr(1, LCase(CStr(ws.Cells(10, c).Value)), LCase(valueLabel)) > 0 Then
            P2_FindNIMCol = c: Exit Function
        End If
    Next c

    P2_FindNIMCol = 0
End Function

Private Function P2_GetNormNIM(qsWb As Workbook, yyyymm As Long) As Double
    On Error GoTo ErrH
    Dim ws As Worksheet
    On Error Resume Next: Set ws = qsWb.Sheets(T_NORM_NIM): On Error GoTo ErrH
    If ws Is Nothing Then P2_GetNormNIM = 0: Exit Function

    Dim nimCol As Long: nimCol = P2_FindNIMCol(ws, yyyymm, "NII")
    If nimCol = 0 Then P2_GetNormNIM = 0: Exit Function

    Dim r As Long
    For r = 228 To 245
        Dim v As Variant: v = ws.Cells(r, nimCol).Value
        If IsNumeric(v) Then
            If CDbl(v) > 0.003 And CDbl(v) < 0.03 Then
                P2_GetNormNIM = CDbl(v): Exit Function
            End If
        End If
    Next r

    P2_GetNormNIM = 0: Exit Function
ErrH: P2_GetNormNIM = 0
End Function

'─── UPDATE DIRECT COST LINKS ──────────────────────────────────────────────────
' Rows 29-30 contain external-link formulas, e.g.:
'   col H (Mar): ='\\server\[Direct Cost - Mar 26 - PF.xls]PF'!$Z$31
'   col I (Apr): ='\\server\[Direct Cost - Apr 26 - PF.xls]PF'!$AA$31
'
' Because the target file (Apr) does not exist yet, reading .Formula from a
' broken-link cell returns garbage — so we never rely on reading prevCol.
' Instead we scan ALL columns in row 29 to find ANY cell with a valid "='"
' external-link formula, use it as a template, work out its month from row 8,
' calculate the column offset to the current month, then build the new formula
' by substituting the month label and advancing the external-file column letter
' by the same offset.

Private Sub P2_UpdateDCLinks(plWs As Worksheet, yyyymm As Long)
    Dim mo      As Long:   mo      = yyyymm Mod 100
    Dim yr2     As String: yr2     = Right(CStr(yyyymm \ 100), 2)
    Dim currLbl As String: currLbl = P2_MonthLabel(yyyymm)
    Dim shortMo As String: shortMo = Left(currLbl, 3)

    ' Declare month name array at top of sub — illegal inside a For loop in VBA
    Dim mNames(1 To 12) As String
    mNames(1)="Jan": mNames(2)="Feb":  mNames(3)="Mar":  mNames(4)="Apr"
    mNames(5)="May": mNames(6)="Jun":  mNames(7)="Jul":  mNames(8)="Aug"
    mNames(9)="Sep": mNames(10)="Oct": mNames(11)="Nov": mNames(12)="Dec"

    ' ── Find currCol: scan row 8 for FIRST match of shortMo ──────────────────
    ' Exit For on first hit — prevents landing on a later YTD/budget col that
    ' also contains the month name (was causing currCol=50).
    Dim currCol As Long: currCol = 0
    Dim c As Long
    For c = 1 To 30
        Dim hdr As String: hdr = Trim(CStr(plWs.Cells(8, c).Value))
        If InStr(1, hdr, shortMo, vbTextCompare) > 0 Then
            currCol = c
            Exit For
        End If
    Next c
    If currCol = 0 Then currCol = 5 + mo
    P2Log "Direct Cost Links", "INFO", "currCol=" & currCol & " (" & shortMo & " " & yr2 & ")"

    ' ── Find template: last valid external-link formula in row 29, col 1..currCol
    Dim tmplF     As String: tmplF     = ""
    Dim tmplCol   As Long:   tmplCol   = 0
    Dim tmplMonth As String: tmplMonth = ""
    Dim tmplYr2   As String: tmplYr2   = ""
    Dim mi As Long, yp As Long, h8 As String, f As String

    For c = 1 To currCol
        f = ""
        On Error Resume Next: f = plWs.Cells(29, c).Formula: On Error GoTo 0
        If Left(f, 2) = "='" And InStr(f, "[") > 0 Then
            tmplF   = f
            tmplCol = c
            h8 = Trim(CStr(plWs.Cells(8, c).Value))
            tmplMonth = ""
            For mi = 1 To 12
                If InStr(1, h8, mNames(mi), vbTextCompare) > 0 Then
                    tmplMonth = mNames(mi): Exit For
                End If
            Next mi
            tmplYr2 = ""
            For yp = 1 To Len(h8) - 1
                If IsNumeric(Mid(h8, yp, 2)) And CLng(Mid(h8, yp, 2)) >= 20 Then
                    tmplYr2 = Mid(h8, yp, 2): Exit For
                End If
            Next yp
        End If
    Next c

    If tmplF = "" Then
        P2Log "Direct Cost Links", "INFO", _
              "No external-link template found in row 29 cols 1-" & currCol & ". Cannot write formulas."
        Exit Sub
    End If

    P2Log "Direct Cost Links", "INFO", _
          "Template col " & tmplCol & " (" & tmplMonth & " " & tmplYr2 & "): " & Left(tmplF, 80)

    Dim colOffset As Long: colOffset = currCol - tmplCol

    Dim savedAlerts As Boolean: savedAlerts = Application.DisplayAlerts
    Dim savedAsk    As Boolean: savedAsk    = Application.AskToUpdateLinks
    Application.DisplayAlerts   = False
    Application.AskToUpdateLinks = False

    ' ── Find a valid row-30 template by scanning cols 1..currCol (same as row 29)
    '    Do this BEFORE the write loop so broken-link cells in row 30 don't
    '    force a fallback to the row-29 template (which has a different row ref).
    Dim tmplF30     As String: tmplF30     = ""
    Dim tmplCol30   As Long:   tmplCol30   = 0
    Dim tmplMonth30 As String: tmplMonth30 = ""
    Dim tmplYr230   As String: tmplYr230   = ""
    Dim f30 As String, h8_30 As String, mi30 As Long, yp30 As Long

    For c = 1 To currCol
        f30 = ""
        On Error Resume Next: f30 = plWs.Cells(30, c).Formula: On Error GoTo 0
        If Left(f30, 2) = "='" And InStr(f30, "[") > 0 Then
            tmplF30   = f30
            tmplCol30 = c
            h8_30 = Trim(CStr(plWs.Cells(8, c).Value))
            tmplMonth30 = ""
            For mi30 = 1 To 12
                If InStr(1, h8_30, mNames(mi30), vbTextCompare) > 0 Then
                    tmplMonth30 = mNames(mi30): Exit For
                End If
            Next mi30
            tmplYr230 = ""
            For yp30 = 1 To Len(h8_30) - 1
                If IsNumeric(Mid(h8_30, yp30, 2)) And CLng(Mid(h8_30, yp30, 2)) >= 20 Then
                    tmplYr230 = Mid(h8_30, yp30, 2): Exit For
                End If
            Next yp30
        End If
    Next c

    P2Log "Direct Cost Links", "INFO", _
          "Row30 template col=" & tmplCol30 & " (" & tmplMonth30 & " " & tmplYr230 & "): " & Left(tmplF30, 80)

    Dim r As Long, rowTmpl As String, newF As String, wErr As Long
    ' ── Build row 29 formula ──────────────────────────────────────────────────
    Dim newF29 As String
    newF29 = tmplF
    If tmplMonth <> "" And tmplYr2 <> "" Then
        newF29 = Replace(newF29, tmplMonth & " " & tmplYr2, shortMo & " " & yr2)
    End If
    newF29 = P2_AdvanceExtColByN(newF29, colOffset)

    wErr = 0
    On Error Resume Next: plWs.Cells(29, currCol).Formula = newF29: wErr = Err.Number: On Error GoTo 0
    If wErr <> 0 Then
        P2Log "Direct Cost Links", "INFO", "Row 29 write error " & wErr & ": " & Left(newF29, 100)
    Else
        P2Log "Direct Cost Links", "OK", "Row 29 col " & currCol & ": " & Left(newF29, 100)
    End If

    ' ── Build row 30 formula ──────────────────────────────────────────────────
    ' Strategy A: use the scanned row-30 template (different row ref from row 29)
    ' Strategy B: fallback — derive from the just-written row-29 formula by
    '             replacing its external-file row number with the row-30 row number
    '             sourced from any valid row-30 formula we can find in other cols.
    Dim newF30 As String

    If tmplF30 <> "" Then
        ' Strategy A — we have a genuine row-30 template
        Dim offset30 As Long: offset30 = currCol - tmplCol30
        newF30 = tmplF30
        If tmplMonth30 <> "" And tmplYr230 <> "" Then
            newF30 = Replace(newF30, tmplMonth30 & " " & tmplYr230, shortMo & " " & yr2)
        End If
        newF30 = P2_AdvanceExtColByN(newF30, offset30)
        P2Log "Direct Cost Links", "INFO", "Row 30: using scanned template (Strategy A)"
    Else
        ' Strategy B — no valid row-30 template found; derive from row-29 new formula
        ' by replacing the external-file row number.  We need to know which $ROW$
        ' the row-29 formula references vs what row 30 should reference.
        ' Read the actual row number from the row-29 template formula (tmplF).
        Dim bangPos30 As Long: bangPos30 = InStrRev(tmplF, "!")
        Dim rowRef29  As String: rowRef29 = ""
        If bangPos30 > 0 Then
            ' After "!" comes $COL$ROW  e.g.  !$Z$31
            Dim afterBang As String: afterBang = Mid(tmplF, bangPos30 + 1)
            ' Find second $ which precedes the row number
            Dim dollar2 As Long: dollar2 = InStr(2, afterBang, "$")
            If dollar2 > 0 Then
                Dim rest30 As String: rest30 = Mid(afterBang, dollar2 + 1)
                Dim rn As Long: rn = 0
                Dim rk As Long
                For rk = 1 To Len(rest30)
                    If IsNumeric(Mid(rest30, rk, 1)) Then
                        rn = rn * 10 + CLng(Mid(rest30, rk, 1))
                    Else
                        Exit For
                    End If
                Next rk
                If rn > 0 Then rowRef29 = CStr(rn)
            End If
        End If

        ' Now read the row number from any existing row-30 formula in other cols
        ' We'll scan cols 1..currCol-1 looking for a valid row-30 formula
        Dim rowRef30 As String: rowRef30 = ""
        Dim fc30 As Long
        For fc30 = 1 To currCol - 1
            Dim ftest As String: ftest = ""
            On Error Resume Next: ftest = plWs.Cells(30, fc30).Formula: On Error GoTo 0
            If Left(ftest, 2) = "='" And InStr(ftest, "!") > 0 Then
                Dim bp As Long: bp = InStrRev(ftest, "!")
                Dim ab30 As String: ab30 = Mid(ftest, bp + 1)
                Dim d2 As Long: d2 = InStr(2, ab30, "$")
                If d2 > 0 Then
                    Dim rr As String: rr = Mid(ab30, d2 + 1)
                    Dim rnum As Long: rnum = 0
                    Dim rki As Long
                    For rki = 1 To Len(rr)
                        If IsNumeric(Mid(rr, rki, 1)) Then
                            rnum = rnum * 10 + CLng(Mid(rr, rki, 1))
                        Else
                            Exit For
                        End If
                    Next rki
                    If rnum > 0 Then rowRef30 = CStr(rnum): Exit For
                End If
            End If
        Next fc30

        ' Build row-30 formula from row-29 new formula, swapping the row reference
        newF30 = newF29
        If rowRef29 <> "" And rowRef30 <> "" And rowRef29 <> rowRef30 Then
            ' Replace "$<row29>  with  $<row30>  after the "!" only
            Dim bangInNew As Long: bangInNew = InStrRev(newF30, "!")
            If bangInNew > 0 Then
                Dim prefix30 As String: prefix30 = Left(newF30, bangInNew)
                Dim suffix30 As String: suffix30 = Mid(newF30, bangInNew + 1)
                suffix30 = Replace(suffix30, "$" & rowRef29, "$" & rowRef30, 1, 1)
                newF30 = prefix30 & suffix30
            End If
            P2Log "Direct Cost Links", "INFO", _
                  "Row 30: derived from row-29 formula; replaced $" & rowRef29 & " with $" & rowRef30 & " (Strategy B)"
        Else
            P2Log "Direct Cost Links", "INFO", _
                  "Row 30: Strategy B row-ref detection failed (r29=" & rowRef29 & " r30=" & rowRef30 & "). Using row-29 formula as-is."
        End If
    End If

    wErr = 0
    On Error Resume Next: plWs.Cells(30, currCol).Formula = newF30: wErr = Err.Number: On Error GoTo 0
    If wErr <> 0 Then
        P2Log "Direct Cost Links", "INFO", "Row 30 write error " & wErr & ": " & Left(newF30, 100)
    Else
        P2Log "Direct Cost Links", "OK", "Row 30 col " & currCol & ": " & Left(newF30, 100)
    End If

    Application.DisplayAlerts   = savedAlerts
    Application.AskToUpdateLinks = savedAsk
End Sub

'─── ADVANCE THE $COL$ AFTER "!" BY N STEPS ───────────────────────────────────
' Used when the template column is not adjacent to the target column.
' e.g. template is col H (Mar, $Z$31), target is col I (Apr, offset=1) → $AA$31
'      template is col G (Feb, $Y$31), target is col I (Apr, offset=2) → $AA$31

Private Function P2_AdvanceExtColByN(f As String, n As Long) As String
    If n = 0 Then P2_AdvanceExtColByN = f: Exit Function

    Dim bangPos As Long: bangPos = InStrRev(f, "!")
    If bangPos = 0 Then P2_AdvanceExtColByN = f: Exit Function

    Dim pos As Long: pos = bangPos + 1
    If pos > Len(f) Or Mid(f, pos, 1) <> "$" Then
        P2_AdvanceExtColByN = f: Exit Function
    End If
    pos = pos + 1   ' skip $, now at first letter

    Dim colStart As Long: colStart = pos
    Do While pos <= Len(f)
        If Mid(f, pos, 1) >= "A" And Mid(f, pos, 1) <= "Z" Then
            pos = pos + 1
        Else
            Exit Do
        End If
    Loop
    Dim colLetters As String: colLetters = Mid(f, colStart, pos - colStart)
    If Len(colLetters) = 0 Then P2_AdvanceExtColByN = f: Exit Function

    ' Convert letters → number, add n, convert back
    Dim num As Long: num = 0
    Dim k As Long
    For k = 1 To Len(colLetters)
        num = num * 26 + (Asc(Mid(colLetters, k, 1)) - Asc("A") + 1)
    Next k
    num = num + n

    Dim newCol As String: newCol = ""
    Dim tmp As Long: tmp = num
    Do While tmp > 0
        newCol = Chr(Asc("A") + (tmp - 1) Mod 26) & newCol
        tmp = (tmp - 1) \ 26
    Loop

    P2_AdvanceExtColByN = Left(f, colStart - 1) & newCol & Mid(f, pos)
End Function

'─── ADVANCE THE $COL$ REFERENCE AFTER "!" IN AN EXTERNAL LINK FORMULA ────────
Private Function P2_AdvanceExternalLinkCol(f As String) As String
    P2_AdvanceExternalLinkCol = P2_AdvanceExtColByN(f, 1)
End Function

'─── ADVANCE A COLUMN LETTER STRING BY ONE ────────────────────────────────────
Private Function P2_NextColLetter(colLetters As String) As String
    Dim n As Long: n = 0
    Dim k As Long
    For k = 1 To Len(colLetters)
        n = n * 26 + (Asc(Mid(colLetters, k, 1)) - Asc("A") + 1)
    Next k
    n = n + 1
    Dim res As String: res = ""
    Do While n > 0
        res = Chr(Asc("A") + (n - 1) Mod 26) & res
        n = (n - 1) \ 26
    Loop
    P2_NextColLetter = res
End Function

'─── BUILD COMMENTARY ──────────────────────────────────────────────────────────
' Fixed column sources (v4):
'   currNIM  = row 14, col N (14)   — current month NIM
'   currEOP  = row 11, col N (14)   — current month EOP asset balance
'   prevNIM  = row 14, dynamic prevCol  — previous month NIM for MoM bps calc
'   prevEOP  = row 11, dynamic prevCol  — previous month EOP for MoM delta
'   currNonNii = row 24, col I (9)  — current month Non-NII
'   prevNonNii = row 24, col H (8)  — previous month Non-NII
'   All YTD figures        = col N (14)
'   vs Budget $            = col R (18)
'   vs Budget %            = col S (19)
'   Budget FY              = col W (23)

Private Function P2_BuildCommentary(plWs As Worksheet, yyyymm As Long, normNIM As Double) As String
    Dim dashWs As Worksheet
    On Error Resume Next
    Set dashWs = plWs.Parent.Sheets(T_PL_DASH)
    On Error GoTo 0
    If dashWs Is Nothing Then Set dashWs = plWs

    Dim mo As Long: mo = yyyymm Mod 100
    Dim shortMo   As String: shortMo   = Left(P2_MonthLabel(yyyymm), 3)
    Dim prevMM    As Long:   prevMM    = P2_PrevYYYYMM(yyyymm)
    Dim shortPrev As String: shortPrev = Left(P2_MonthLabel(prevMM), 3)

    ' Dynamic column scan — still needed for prevNIM, prevEOP MoM delta,
    ' headcount col, and any other dynamic-month references
    Dim currCol As Long: currCol = 0
    Dim prevCol As Long: prevCol = 0
    Dim c As Long
    For c = 1 To 30
        Dim hdr As String: hdr = Trim(CStr(dashWs.Cells(8, c).Value))
        If InStr(1, hdr, shortMo, vbTextCompare) > 0 Then currCol = c
        If InStr(1, hdr, shortPrev, vbTextCompare) > 0 Then prevCol = c
    Next c
    If currCol = 0 Then currCol = 5 + mo
    If prevCol = 0 Then prevCol = currCol - 1

    Dim mLbl As String: mLbl = P2_MonthLabel(yyyymm)
    Dim pLbl As String: pLbl = P2_MonthLabel(prevMM)

    ' Fixed column indices
    Const CN As Long = 14   ' Col N — current month actual / YTD actual
    Const CH As Long = 8    ' Col H — previous month Non-NII (row 24)
    Const CI As Long = 9    ' Col I — current month Non-NII (row 24)
    Const CR As Long = 18   ' Col R — vs Budget $
    Const CS As Long = 19   ' Col S — vs Budget %
    Const CW As Long = 23   ' Col W — Budget FY

    ' ── YTD P&L ──────────────────────────────────────────────────────────────
    Dim totIncYTD As Double: totIncYTD = P2V(dashWs, 25, CN)
    Dim totVsBgtD As Double: totVsBgtD = P2V(dashWs, 25, CR)
    Dim totVsBgtP As Double: totVsBgtP = P2V(dashWs, 25, CS)
    Dim niiYTD    As Double: niiYTD    = P2V(dashWs, 23, CN)
    Dim niiVsBgtD As Double: niiVsBgtD = P2V(dashWs, 23, CR)
    Dim nonNiiYTD As Double: nonNiiYTD = P2V(dashWs, 24, CN)
    Dim nonNiiVsD As Double: nonNiiVsD = P2V(dashWs, 24, CR)

    ' ── NIM: bracketNIM = YTD NIM from col N (14) row 14 — for compressed NIM bracket in NII section
    '         currNIM   = current month NIM from col I (9)  row 14 — for NIM MoM sentence only
    '         prevNIM   = previous month NIM from dynamic prevCol row 14 — for MoM bps calc
    Dim currNIM    As Double: currNIM    = P2V(dashWs, 14, CI)       ' col I row 14 — MoM sentence only
    Dim bracketNIM As Double: bracketNIM = P2V(dashWs, 14, CN)       ' col N row 14 — YTD NIM for compressed NIM bracket (NOT col I)
    Dim prevNIM    As Double: prevNIM    = P2V(dashWs, 14, prevCol)  ' MoM prev — dynamic
    Dim bgtNIM  As Double: bgtNIM  = P2V(dashWs, 14, CW)
    Dim nimBps  As Long:   nimBps  = CLng((currNIM - prevNIM) * 100)

    ' ── EOP Assets: current month fixed to col N (14); prev month dynamic ─────
    Dim currEOP As Double: currEOP = P2V(dashWs, 11, CN)       ' FIX v4: col N row 11
    Dim prevEOP As Double: prevEOP = P2V(dashWs, 11, prevCol)  ' MoM prev — dynamic
    Dim eopMoM  As Double: eopMoM  = currEOP - prevEOP
    Dim eopVsD  As Double: eopVsD  = P2V(dashWs, 11, CR)

    ' ── YTD avg asset and budget (for NII bracket) ────────────────────────────
    Dim ytdEOP  As Double: ytdEOP  = P2V(dashWs, 11, CN)   ' row 11 col N = YTD actual
    Dim bgtEOP  As Double: bgtEOP  = P2V(dashWs, 11, CW)   ' row 11 col W = budget

    ' ── Non-NII MoM: col I (9) = current month; col H (8) = previous month ───
    Dim currNonNii As Double: currNonNii = P2V(dashWs, 24, CI)  ' FIX v4: col I row 24
    Dim prevNonNii As Double: prevNonNii = P2V(dashWs, 24, CH)  ' FIX v4: col H row 24
    Dim nonNiiMoM  As Double: nonNiiMoM  = currNonNii - prevNonNii

    ' ── Cross-selling ─────────────────────────────────────────────────────────
    Dim xsellYTD As Double: xsellYTD = P2V(dashWs, 72, CN)
    Dim cashYTD  As Double: cashYTD  = P2V(dashWs, 76, CN)
    Dim gfmYTD   As Double: gfmYTD   = P2V(dashWs, 74, CN)
    Dim tradeYTD As Double: tradeYTD = P2V(dashWs, 75, CN)

    ' ── Direct Expense ────────────────────────────────────────────────────────
    Dim dirExpYTD As Double: dirExpYTD = P2V(dashWs, 31, CN)
    Dim dirExpVsD As Double: dirExpVsD = P2V(dashWs, 31, CR)

    ' ── Headcount ─────────────────────────────────────────────────────────────
    Dim hc      As Long: hc      = CLng(P2V(dashWs, 68, currCol))
    Dim hcBgt   As Long: hcBgt   = CLng(P2V(dashWs, 68, 15))
    Dim hcVsBgt As Long: hcVsBgt = hc - hcBgt

    ' ── Build text ────────────────────────────────────────────────────────────
    Dim t As String

    ' Section 1: YTD vs Budget header
    t = "YTD Act Vs Budget" & vbNewLine
    t = t & "YTD " & mLbl & " Total Income of S" & FM(Abs(totIncYTD)) & " was " & _
        FM(Abs(totVsBgtD)) & " (" & Format(Abs(totVsBgtP) * 100, "0") & "%) " & _
        AB(totVsBgtD) & " budget, driven primarily by " & SW(niiVsBgtD) & " NII (" & _
        FMS(niiVsBgtD) & ") offset by " & SW(nonNiiVsD) & " non-interest income of " & _
        FMS(nonNiiVsD) & "." & vbNewLine & vbNewLine

    ' Section 2: NII — bracket shows YTD avg asset (col N row 11) vs budget,
    '            and current month NIM (col N row 14) vs budget NIM (col W row 14)
    t = t & "NII of " & FM(Abs(niiYTD)) & " was " & FM(Abs(niiVsBgtD)) & " " & _
        AB(niiVsBgtD) & " budget due to [UPDATE: reason e.g. lower avg asset volume " & _
        "(" & FB(Abs(ytdEOP)) & " vs " & FB(Abs(bgtEOP)) & " budget) and compressed NIM " & _
        "(" & Format(Abs(bracketNIM), "0.00") & "% vs " & Format(Abs(bgtNIM), "0.00") & "% budget)]." & _
        vbNewLine & vbNewLine

    ' Section 3: NIM MoM — curr = col N row 14; prev = dynamic prevCol row 14
    t = t & "NIM " & ID(nimBps) & " " & Abs(nimBps) & "bps MoM (from " & _
        Format(Abs(prevNIM), "0.00") & "% in " & pLbl & " to " & _
        Format(Abs(currNIM), "0.00") & "% in " & mLbl & ") due to [UPDATE: reason]. "
    If normNIM > 0 Then
        Dim normNIMDisplay As Double
        normNIMDisplay = IIf(normNIM < 0.1, normNIM * 100, normNIM)
        t = t & "Normalised NIM maintains at " & Format(normNIMDisplay, "0.00") & "%."
    End If
    t = t & vbNewLine & vbNewLine

    ' Section 4: EOP Assets MoM — curr = col N row 11; prev = dynamic prevCol row 11
    t = t & "EOP Asset Balance " & GD(eopMoM) & " " & FM(Abs(eopMoM)) & " MoM to S" & _
        FB(Abs(currEOP)) & ", but remained " & FM(Abs(eopVsD)) & " " & AB(eopVsD) & _
        " budget (budget: " & FB(Abs(bgtEOP)) & ") resulting in a " & _
        IIf(eopVsD < 0, "drag on", "support for") & " average asset volume." & vbNewLine & vbNewLine

    ' Section 5: Non-NII MoM — I24 minus H24; display total = I24
    '            Pass currNonNii (I24) and prevNonNii (H24) as coverage denominators
    Dim feeWs As Worksheet
    On Error Resume Next: Set feeWs = plWs.Parent.Sheets(T_PF_FEES): On Error GoTo 0
    Dim feeText As String
    feeText = P2_GetFeeContributors(feeWs, yyyymm, currNonNii, prevNonNii)

    t = t & mLbl & " Non-NII " & GD(nonNiiMoM) & " " & FM(Abs(nonNiiMoM)) & _
        " MoM to " & FM(Abs(currNonNii)) & ". Key fee contributors:" & _
        vbNewLine & vbNewLine
    t = t & feeText

    ' Section 6: X-selling — no extra blank line; feeText already ends with vbNewLine
    t = t & "YTD " & mLbl & " X-selling Income of " & FM(Abs(xsellYTD)) & _
        " is mainly derived from:" & vbNewLine
    t = t & "- Cash +" & Format(Abs(cashYTD) / 1000000, "0.0") & "m" & vbNewLine
    t = t & "- GFM Sales $" & Format(Abs(gfmYTD) / 1000000, "0.0") & "m; and" & vbNewLine
    t = t & "- Trade +" & Format(Abs(tradeYTD) / 1000000, "0.0") & "m" & vbNewLine & vbNewLine

    ' Section 7: Direct Expense
    t = t & "YTD " & mLbl & " Direct Expense of " & FM(Abs(dirExpYTD)) & _
        " was " & FM(Abs(dirExpVsD)) & " " & AB(dirExpVsD) & " budget." & _
        vbNewLine & vbNewLine

    ' Section 8: Headcount
    t = t & mLbl & " Regional Headcount remained at " & hc & _
        "; " & Format(hcVsBgt, "+0;-0") & " vs Bgt."

    P2_BuildCommentary = t
End Function

'─── UPDATE COMMENTARY TEXT BOX ────────────────────────────────────────────────

Private Sub P2_UpdateTextBox(dashWs As Worksheet, commentary As String)
    Dim shp As Shape, target As Shape

    For Each shp In dashWs.Shapes
        If shp.Type = msoTextBox Then
            If shp.Left >= dashWs.Columns("Z").Left Then
                Set target = shp: Exit For
            End If
        End If
    Next shp

    If target Is Nothing Then
        For Each shp In dashWs.Shapes
            If shp.Type = msoTextBox Then Set target = shp: Exit For
        Next shp
    End If

    If Not target Is Nothing Then
        With target.TextFrame
            .AutoSize = False
            .Characters.Text = commentary
            With .Characters.Font
                .Bold = False
                .Underline = xlUnderlineStyleNone
                .Size = 9
                .Name = "Calibri"
            End With
            Dim lines() As String: lines = Split(commentary, vbNewLine)
            Dim charPos As Long: charPos = 1
            Dim i As Long
            For i = 0 To UBound(lines)
                Dim lineLen As Long: lineLen = Len(lines(i))
                If lineLen > 0 Then
                    If lines(i) = "YTD Act Vs Budget" Or _
                       Left(Trim(lines(i)), 6) = "Major " Then
                        .Characters(charPos, lineLen).Font.Bold = True
                    End If
                End If
                charPos = charPos + lineLen + 1
            Next i
        End With
    Else
        MsgBox "Commentary text box not found on " & T_PL_DASH & "." & vbNewLine & _
               "Please paste commentary manually.", vbExclamation
    End If
End Sub

'─── CREATE SENT FILE ──────────────────────────────────────────────────────────
' FIX v3: Activate dashboard FIRST (before any hiding) so Excel always has at
'         least one visible sheet and never blocks the hide operations.
'         All tabs to hide are collected into one array and processed together.

Private Sub P2_CreateSentFile(pfWb As Workbook, yyyymm As Long, sentPath As String)
    Dim tempPath As String
    tempPath = Left(sentPath, Len(sentPath) - 5) & "_temp.xlsm"
    pfWb.SaveCopyAs tempPath

    Dim sentWb As Workbook
    On Error Resume Next
    Set sentWb = Workbooks.Open(tempPath)
    On Error GoTo 0
    If sentWb Is Nothing Then
        On Error Resume Next: Kill tempPath: On Error GoTo 0
        Exit Sub
    End If

    Application.ScreenUpdating = False

    ' 1. Paste values + formats on all key tabs
    P2_PasteValues sentWb, T_PL_PF
    P2_PasteValues sentWb, T_PL_DASH
    P2_PasteValues sentWb, T_PF_FEES
    P2_PasteValues sentWb, T_INC_CNTRY
    P2_PasteValues sentWb, T_XSELL
    P2_PasteValues sentWb, T_CUST_LIST

    ' 2. Format visible pivot tabs
    P2_FormatSentTab sentWb, T_PF_FEES
    P2_FormatSentTab sentWb, T_XSELL
    P2_FormatSentTab sentWb, T_CUST_LIST

    ' 3. Close all groups
    Dim ws As Worksheet
    For Each ws In sentWb.Sheets
        If ws.Visible = xlSheetVisible Then
            On Error Resume Next
            ws.Outline.ShowLevels RowLevels:=1, ColumnLevels:=1
            On Error GoTo 0
        End If
    Next ws

    ' 4. Break all external links
    On Error Resume Next
    Dim links As Variant: links = sentWb.LinkSources(xlExcelLinks)
    If Not IsEmpty(links) Then
        Dim i As Long
        For i = 1 To UBound(links)
            sentWb.BreakLink Name:=links(i), Type:=xlLinkTypeExcelLinks
        Next i
    End If
    Application.Calculate
    On Error GoTo 0

    ' 5. ACTIVATE DASHBOARD FIRST — must be done before hiding any sheets
    '    so Excel always has a visible active sheet throughout the hide loop
    On Error Resume Next
    sentWb.Sheets(T_PL_DASH).Visible = xlSheetVisible
    sentWb.Sheets(T_PL_DASH).Activate
    On Error GoTo 0

    ' 6. Hide tabs — all in one consolidated list
    Dim hideList As Variant
    hideList = Array( _
        T_PL_PF, _
        T_PL_XCROSS, _
        T_WORK_PF, _
        T_WORKING, _
        T_TO_UPDATE, _
        T_WORK_PIVOTS, _
        T_NORM_NIM _
    )

    Dim h As Long
    For h = 0 To UBound(hideList)
        Dim tabName As String: tabName = CStr(hideList(h))
        On Error Resume Next
        Dim wsH As Worksheet: Set wsH = sentWb.Sheets(tabName)
        If Not wsH Is Nothing Then
            If wsH.Name <> sentWb.ActiveSheet.Name Then
                wsH.Visible = xlSheetHidden
            End If
        End If
        Set wsH = Nothing
        On Error GoTo 0
    Next h

    ' 7. Delete all cell notes/comments from every sheet before saving
    Dim wsCN As Worksheet
    For Each wsCN In sentWb.Sheets
        On Error Resume Next
        wsCN.Cells.ClearComments
        On Error GoTo 0
    Next wsCN

    ' 8. Save as xlsx (strips macros) and delete temp xlsm
    Application.DisplayAlerts = False
    On Error Resume Next
    sentWb.SaveAs sentPath, xlOpenXMLWorkbook
    sentWb.Close
    Kill tempPath
    On Error GoTo 0
    Application.DisplayAlerts = True
End Sub

Private Sub P2_PasteValues(wb As Workbook, tabName As String)
    On Error Resume Next
    Dim ws As Worksheet: Set ws = wb.Sheets(tabName): On Error GoTo 0
    If ws Is Nothing Then Exit Sub
    Dim rng As Range: Set rng = ws.UsedRange
    rng.Copy
    rng.PasteSpecial xlPasteValues
    rng.PasteSpecial xlPasteFormats
    Application.CutCopyMode = False
    ws.Cells.WrapText = False
End Sub

'─── FEE CONTRIBUTORS FROM PF FEES TAB ────────────────────────────────────────
' v4 CHANGES:
'   - Signature: added currTotal (= dashboard I24) and prevTotal (= dashboard H24)
'     as the denominators for 80% cumulative coverage calculation.
'   - Coverage logic: contributors are listed in descending absolute value order
'     until their running sum of absolute values >= 80% of Abs(currTotal) for the
'     current month block, and >= 80% of Abs(prevTotal) for the previous month
'     block.  If the total passed in is zero, falls back to listing all entries.
'   - All scanning / logging / fallback logic is unchanged from v3.
'
' Output format:
'   Major Apr 26 Fees:
'   [blank line]
'   1) Customer A $x.xm
'   2) Customer B $x.xm
'   [blank line]
'   Major Mar 26 Fees:
'   [blank line]
'   1) Customer C $x.xm
'   [blank line]
'   ← caller adds vbNewLine before "YTD X-selling" line

Private Function P2_GetFeeContributors( _
        feeWs As Worksheet, _
        yyyymm As Long, _
        currTotal As Double, _
        prevTotal As Double) As String

    On Error GoTo ErrH

    If feeWs Is Nothing Then
        P2_GetFeeContributors = "Major " & P2_MonthLabel(yyyymm) & " Fees:" & vbNewLine & _
                                "[UPDATE: PF Fees tab not found]" & vbNewLine
        Exit Function
    End If

    Dim prevMM  As Long:   prevMM  = P2_PrevYYYYMM(yyyymm)
    Dim currLbl As String: currLbl = P2_MonthLabel(yyyymm)
    Dim prevLbl As String: prevLbl = P2_MonthLabel(prevMM)

    ' ── Read entire used range into array ─────────────────────────────────────
    Dim pvData As Variant
    pvData = feeWs.UsedRange.Value   ' 1-based 2D array, preserves true VBA types

    Dim nRows As Long: nRows = UBound(pvData, 1)
    Dim nCols As Long: nCols = UBound(pvData, 2)

    P2Log "PF Fees Search", "INFO", "UsedRange array: " & nRows & " rows x " & nCols & " cols"

    ' ── Scan ALL rows for YYYYMM header ───────────────────────────────────────
    Dim hdRow   As Long: hdRow   = 0
    Dim currCol As Long: currCol = 0
    Dim prevCol As Long: prevCol = 0
    Dim r As Long, c As Long
    Dim diagLog As String
    diagLog = "Fee array scan — looking for " & yyyymm & " and " & prevMM & ":" & vbNewLine

    For r = 1 To nRows
        For c = 1 To nCols
            Dim cv As Variant: cv = pvData(r, c)
            If IsEmpty(cv) Then GoTo NextArrCell

            Dim cvLong As Long: cvLong = 0

            If IsNumeric(cv) Then
                Dim cvD As Double: cvD = CDbl(cv)
                ' Excel date serial range for 2020-2035
                If cvD >= 43831 And cvD <= 49673 Then
                    On Error Resume Next
                    Dim dt As Date: dt = CDate(cvD)
                    If Err.Number = 0 Then cvLong = Year(dt) * 100 + Month(dt)
                    Err.Clear
                    On Error GoTo ErrH
                ' Direct YYYYMM integer (200001 to 209912)
                ElseIf cvD >= 200001 And cvD <= 209912 Then
                    cvLong = CLng(cvD)
                End If
            Else
                ' Text — try parsing as 6-digit YYYYMM
                Dim cvStr As String: cvStr = Trim(CStr(cv))
                If Len(cvStr) = 6 And IsNumeric(cvStr) Then
                    cvLong = CLng(cvStr)
                End If
            End If

            If cvLong > 0 Then
                diagLog = diagLog & "  R" & r & "C" & c & "=" & cvLong
                If cvLong = yyyymm Then
                    currCol = c
                    If hdRow = 0 Then hdRow = r
                    diagLog = diagLog & " <<CURR"
                ElseIf cvLong = prevMM Then
                    prevCol = c
                    If hdRow = 0 Then hdRow = r
                    diagLog = diagLog & " <<PREV"
                End If
                diagLog = diagLog & vbNewLine
            End If

NextArrCell:
        Next c
        If currCol > 0 And prevCol > 0 Then Exit For
    Next r

    P2Log "PF Fees Search", "INFO", diagLog

    ' ── Fallback: raw CStr scan on rows 1-10 ──────────────────────────────────
    If currCol = 0 Then
        Dim yyyymmStr As String: yyyymmStr = CStr(yyyymm)
        Dim prevMMStr As String: prevMMStr = CStr(prevMM)
        For r = 1 To Application.Min(10, nRows)
            For c = 1 To nCols
                Dim rawStr As String: rawStr = Trim(CStr(pvData(r, c)))
                If rawStr = yyyymmStr Then
                    currCol = c
                    If hdRow = 0 Then hdRow = r
                    P2Log "PF Fees Search", "INFO", "Fallback CStr match: R" & r & "C" & c & "=" & rawStr & " <<CURR"
                End If
                If rawStr = prevMMStr Then
                    prevCol = c
                    If hdRow = 0 Then hdRow = r
                    P2Log "PF Fees Search", "INFO", "Fallback CStr match: R" & r & "C" & c & "=" & rawStr & " <<PREV"
                End If
            Next c
        Next r
    End If

    If currCol = 0 Then
        P2Log "PF Fees Search", "INFO", _
              "YYYYMM " & yyyymm & " not found anywhere in PF Fees (" & nRows & "x" & nCols & "). " & _
              "Check Log for all candidate cells found."
        P2_GetFeeContributors = "Major " & currLbl & " Fees: [" & yyyymm & _
                                " column not found — check Log sheet]" & vbNewLine
        Exit Function
    End If

    P2Log "PF Fees Search", "OK", _
          "Found hdRow=" & hdRow & " currCol=" & currCol & _
          " (" & yyyymm & ") prevCol=" & prevCol & " (" & prevMM & ")"

    ' ── Collect customer names and fee values from array ──────────────────────
    ' Customer name from column 1 (col A); skip blank / total / grand total rows
    Dim cNames() As String, cVals() As Double, cCnt As Long: cCnt = 0
    Dim pNames() As String, pVals() As Double, pCnt As Long: pCnt = 0

    For r = hdRow + 1 To nRows
        Dim custN As String
        custN = Trim(CStr(pvData(r, 1)))
        If custN = "" And nCols >= 2 Then custN = Trim(CStr(pvData(r, 2)))
        If custN = "" Then GoTo NextFeeRow
        If InStr(1, LCase(custN), "total") > 0 Then GoTo NextFeeRow
        If InStr(1, LCase(custN), "grand") > 0 Then GoTo NextFeeRow

        ' Current month
        Dim cVal As Double: cVal = 0
        If IsNumeric(pvData(r, currCol)) And Not IsEmpty(pvData(r, currCol)) Then
            cVal = CDbl(pvData(r, currCol))
        End If
        If Abs(cVal) > 0 Then
            cCnt = cCnt + 1
            ReDim Preserve cNames(1 To cCnt): ReDim Preserve cVals(1 To cCnt)
            cNames(cCnt) = custN: cVals(cCnt) = cVal
        End If

        ' Previous month
        If prevCol > 0 Then
            Dim pVal As Double: pVal = 0
            If IsNumeric(pvData(r, prevCol)) And Not IsEmpty(pvData(r, prevCol)) Then
                pVal = CDbl(pvData(r, prevCol))
            End If
            If Abs(pVal) > 0 Then
                pCnt = pCnt + 1
                ReDim Preserve pNames(1 To pCnt): ReDim Preserve pVals(1 To pCnt)
                pNames(pCnt) = custN: pVals(pCnt) = pVal
            End If
        End If

NextFeeRow:
    Next r

    P2Log "PF Fees Search", "INFO", _
          "Data rows scanned: " & hdRow + 1 & " to " & nRows & _
          ". Curr entries=" & cCnt & " Prev entries=" & pCnt

    ' ── Sort both arrays descending by absolute value (bubble sort) ───────────
    Dim ii As Long, jj As Long, tn As String, tv As Double
    If cCnt > 1 Then
        For ii = 1 To cCnt - 1
            For jj = 1 To cCnt - ii
                If Abs(cVals(jj)) < Abs(cVals(jj + 1)) Then
                    tn = cNames(jj): tv = cVals(jj)
                    cNames(jj) = cNames(jj + 1): cVals(jj) = cVals(jj + 1)
                    cNames(jj + 1) = tn: cVals(jj + 1) = tv
                End If
            Next jj
        Next ii
    End If
    If pCnt > 1 Then
        For ii = 1 To pCnt - 1
            For jj = 1 To pCnt - ii
                If Abs(pVals(jj)) < Abs(pVals(jj + 1)) Then
                    tn = pNames(jj): tv = pVals(jj)
                    pNames(jj) = pNames(jj + 1): pVals(jj) = pVals(jj + 1)
                    pNames(jj + 1) = tn: pVals(jj + 1) = tv
                End If
            Next jj
        Next ii
    End If

    ' ── 80% coverage threshold ────────────────────────────────────────────────
    ' Use Abs(currTotal) / Abs(prevTotal) as denominator.
    ' If the passed-in total is zero (e.g. cell was empty), fall back to sum of
    ' all collected entries so that all contributors are shown rather than none.
    Dim cDenom As Double: cDenom = Abs(currTotal)
    If cDenom = 0 Then
        Dim ci2 As Long
        For ci2 = 1 To cCnt: cDenom = cDenom + Abs(cVals(ci2)): Next ci2
    End If

    Dim pDenom As Double: pDenom = Abs(prevTotal)
    If pDenom = 0 Then
        Dim pi2 As Long
        For pi2 = 1 To pCnt: pDenom = pDenom + Abs(pVals(pi2)): Next pi2
    End If

    Const COVERAGE_PCT As Double = 0.8   ' 80%

    ' ── Build result ──────────────────────────────────────────────────────────
    Dim result As String

    ' Current month fees — list until 80% of cDenom covered
    result = "Major " & currLbl & " Fees:" & vbNewLine & vbNewLine
    If cCnt = 0 Then
        result = result & "[No fee entries found for " & currLbl & "]" & vbNewLine
    Else
        Dim cRunning As Double: cRunning = 0
        For ii = 1 To cCnt
            result = result & ii & ") " & cNames(ii) & " $" & _
                     Format(Abs(cVals(ii)) / 1000000, "0.0") & "m" & vbNewLine
            cRunning = cRunning + Abs(cVals(ii))
            If cDenom > 0 And (cRunning / cDenom) >= COVERAGE_PCT Then Exit For
        Next ii
    End If

    ' Previous month fees — list until 80% of pDenom covered
    If prevCol > 0 Then
        result = result & vbNewLine & "Major " & prevLbl & " Fees:" & vbNewLine & vbNewLine
        If pCnt = 0 Then
            result = result & "[No fee entries found for " & prevLbl & "]" & vbNewLine
        Else
            Dim pRunning As Double: pRunning = 0
            For ii = 1 To pCnt
                result = result & ii & ") " & pNames(ii) & " $" & _
                         Format(Abs(pVals(ii)) / 1000000, "0.0") & "m" & vbNewLine
                pRunning = pRunning + Abs(pVals(ii))
                If pDenom > 0 And (pRunning / pDenom) >= COVERAGE_PCT Then Exit For
            Next ii
        End If
    End If

    ' Trailing blank line — caller's vbNewLine before "YTD X-selling" adds the
    ' blank separator between fee block and cross-selling section
    result = result & vbNewLine

    P2Log "PF Fees Search", "OK", _
          "Fee contributors written. Curr covered=" & Format(IIf(cDenom > 0, cRunning / cDenom, 0), "0%") & _
          " Prev covered=" & Format(IIf(pDenom > 0, pRunning / pDenom, 0), "0%")

    P2_GetFeeContributors = result
    Exit Function
ErrH:
    P2_GetFeeContributors = "Major Fees: [error reading PF Fees — " & Err.Description & "]" & vbNewLine
End Function

'─── FORMAT SENT TABS ──────────────────────────────────────────────────────────
' v3 FIX retained: Number format #,##0.00 (2 decimal places)

Private Sub P2_FormatSentTab(wb As Workbook, tabName As String)
    On Error Resume Next
    Dim ws As Worksheet: Set ws = wb.Sheets(tabName)
    On Error GoTo 0
    If ws Is Nothing Then Exit Sub

    ws.Rows("1:3").Delete

    Dim lastRow As Long: lastRow = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    Dim lastCol As Long: lastCol = 1
    Dim r As Long, c As Long
    For r = 1 To Application.Min(10, lastRow)
        Dim rc As Long: rc = ws.Cells(r, ws.Columns.Count).End(xlToLeft).Column
        If rc > lastCol Then lastCol = rc
    Next r

    ' Row 1: light blue header
    With ws.Range(ws.Cells(1, 1), ws.Cells(1, lastCol))
        .Font.Bold = True
        .Font.Color = RGB(0, 0, 0)
        .Interior.Color = RGB(192, 230, 245)
    End With

    ' Row 2: red header, white bold
    With ws.Range(ws.Cells(2, 1), ws.Cells(2, lastCol))
        .Font.Bold = True
        .Font.Color = RGB(255, 255, 255)
        .Interior.Color = RGB(192, 0, 0)
    End With

    ' Bold cols A and B
    ws.Range(ws.Cells(1, 1), ws.Cells(lastRow, 2)).Font.Bold = True

    ' Bold total rows
    For r = 3 To lastRow
        For c = 1 To lastCol
            If Not IsEmpty(ws.Cells(r, c).Value) Then
                If InStr(1, LCase(CStr(ws.Cells(r, c).Value)), "total") > 0 Then
                    ws.Rows(r).Font.Bold = True: Exit For
                End If
            End If
        Next c
    Next r

    ws.Cells.WrapText = False
    For c = 1 To lastCol
        ws.Columns(c).AutoFit
        If ws.Columns(c).ColumnWidth < 12 Then ws.Columns(c).ColumnWidth = 12
    Next c

    ' 2 decimal places
    Dim numRng As Range
    On Error Resume Next
    Set numRng = ws.Range(ws.Cells(3, 4), ws.Cells(lastRow, lastCol))
    If Not numRng Is Nothing Then
        numRng.NumberFormat = "#,##0.00"
    End If
    On Error GoTo 0
End Sub

'─── LOG ───────────────────────────────────────────────────────────────────────

Private Sub P2Log(stepName As String, status As String, msg As String)
    Dim ws As Worksheet
    On Error Resume Next
    Set ws = ThisWorkbook.Sheets("Log")
    On Error GoTo 0

    If ws Is Nothing Then
        Set ws = ThisWorkbook.Sheets.Add(After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
        ws.Name = "Log"
        ws.Cells(1, 1).Value = "Timestamp":  ws.Cells(1, 2).Value = "Source"
        ws.Cells(1, 3).Value = "Step":       ws.Cells(1, 4).Value = "Status"
        ws.Cells(1, 5).Value = "Message"
        ws.Rows(1).Font.Bold = True
        ws.Columns("A").ColumnWidth = 22: ws.Columns("B").ColumnWidth = 8
        ws.Columns("C").ColumnWidth = 30: ws.Columns("D").ColumnWidth = 10
        ws.Columns("E").ColumnWidth = 80
    End If

    ws.Activate

    Dim nr As Long: nr = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row + 1
    ws.Cells(nr, 1).Value = Format(Now, "DD/MM/YYYY HH:MM:SS")
    ws.Cells(nr, 2).Value = "Part2"
    ws.Cells(nr, 3).Value = stepName
    ws.Cells(nr, 4).Value = status
    ws.Cells(nr, 5).Value = msg

    Select Case UCase(status)
        Case "RUNNING": ws.Cells(nr, 4).Interior.Color = RGB(255, 255, 180)
        Case "OK":      ws.Cells(nr, 4).Interior.Color = RGB(180, 255, 180)
        Case "ERROR":   ws.Cells(nr, 4).Interior.Color = RGB(255, 180, 180)
        Case "INFO":    ws.Cells(nr, 4).Interior.Color = RGB(180, 220, 255)
    End Select

    ws.Cells(nr, 1).Select
    Application.ScreenUpdating = True
    DoEvents
End Sub

Private Function P2V(ws As Worksheet, r As Long, c As Long) As Double
    On Error Resume Next
    Dim v As Variant: v = ws.Cells(r, c).Value
    If IsNumeric(v) Then P2V = CDbl(v) Else P2V = 0
End Function

Private Function FM(v As Double) As String
    FM = "$" & Format(Abs(v) / 1000000, "0.0") & "m"
End Function
Private Function FMS(v As Double) As String
    If v >= 0 Then FMS = "+$" & Format(v / 1000000, "0.0") & "m" Else FMS = "-$" & Format(Abs(v) / 1000000, "0.0") & "m"
End Function
Private Function FB(v As Double) As String
    FB = "$" & Format(Abs(v) / 1000000000, "0.0") & "b"
End Function
Private Function AB(v As Double) As String:  AB  = IIf(v >= 0, "above", "below"):            End Function
Private Function SW(v As Double) As String:  SW  = IIf(v >= 0, "stronger", "weaker"):        End Function
Private Function ID(v As Long) As String:    ID  = IIf(v >= 0, "increased", "decreased"):    End Function
Private Function GD(v As Double) As String:  GD  = IIf(v >= 0, "grew", "declined"):          End Function

Private Function P2_ColLetter(col As Long) As String
    If col <= 26 Then
        P2_ColLetter = Chr(64 + col)
    Else
        P2_ColLetter = Chr(64 + (col - 1) \ 26) & Chr(65 + (col - 1) Mod 26)
    End If
End Function

Private Function P2_MonthLabel(yyyymm As Long) As String
    Dim m(1 To 12) As String
    m(1)="Jan": m(2)="Feb": m(3)="Mar": m(4)="Apr":  m(5)="May":  m(6)="Jun"
    m(7)="Jul": m(8)="Aug": m(9)="Sep": m(10)="Oct": m(11)="Nov": m(12)="Dec"
    P2_MonthLabel = m(yyyymm Mod 100) & " " & Right(CStr(yyyymm \ 100), 2)
End Function

Private Function P2_PrevYYYYMM(yyyymm As Long) As Long
    Dim yr As Long: yr = yyyymm \ 100
    Dim mo As Long: mo = yyyymm Mod 100
    If mo = 1 Then P2_PrevYYYYMM = (yr - 1) * 100 + 12 _
    Else P2_PrevYYYYMM = yr * 100 + (mo - 1)
End Function

Private Function P2_NextYYYYMM(yyyymm As Long) As Long
    Dim yr As Long: yr = yyyymm \ 100
    Dim mo As Long: mo = yyyymm Mod 100
    If mo = 12 Then P2_NextYYYYMM = (yr + 1) * 100 + 1 _
    Else P2_NextYYYYMM = yr * 100 + (mo + 1)
End Function

Private Function P2_ExtractPeriod(fileName As String) As Long
    Dim months(1 To 12) As String
    months(1)="Jan": months(2)="Feb": months(3)="Mar": months(4)="Apr"
    months(5)="May": months(6)="Jun": months(7)="Jul": months(8)="Aug"
    months(9)="Sep": months(10)="Oct": months(11)="Nov": months(12)="Dec"
    Dim i As Long
    For i = 1 To 12
        Dim pos As Long: pos = InStr(1, fileName, " " & months(i) & " ", vbTextCompare)
        If pos > 0 Then
            Dim yrStr As String: yrStr = Mid(fileName, pos + Len(months(i)) + 2, 2)
            If IsNumeric(yrStr) Then
                P2_ExtractPeriod = (2000 + CLng(yrStr)) * 100 + i
                Exit Function
            End If
        End If
    Next i
    P2_ExtractPeriod = 0
End Function
