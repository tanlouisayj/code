Option Explicit

Private WithEvents olReminders As Outlook.Reminders

' When Outlook opens: hook the reminders and catch up on today's sweep and the digest if they haven't run
Private Sub Application_Startup()
    Set olReminders = Application.Reminders
    SweepIfDue
    DigestIfDue
End Sub

' The daily/weekly calendar reminders trigger the macros
Private Sub Application_Reminder(ByVal Item As Object)
    On Error Resume Next
    Select Case Item.Subject
        Case SWEEP_SUBJECT
            DoSweep True
        Case DIGEST_SUBJECT
            DigestIfDue
    End Select
End Sub

' Hide the reminder pop-ups for these two so they don't interrupt you
Private Sub olReminders_BeforeReminderShow(Cancel As Boolean)
    Dim r As Outlook.Reminder
    On Error Resume Next
    For Each r In olReminders
        If r.IsVisible Then
            If r.Caption = SWEEP_SUBJECT Or r.Caption = DIGEST_SUBJECT Then r.Dismiss
        End If
    Next r
End Sub
